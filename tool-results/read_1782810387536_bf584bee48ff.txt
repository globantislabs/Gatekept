  2286→function exportToCSV(data: any[], filename: string, columns: { key: string; label: string }[]) {
  2287→  const header = columns.map(c => c.label).join(',')
  2288→  const rows = data.map(row => columns.map(c => {
  2289→    const val = row[c.key]
  2290→    const str = val === null || val === undefined ? '' : String(val)
  2291→    return `"${str.replace(/"/g, '""')}"`
  2292→  }).join(','))
  2293→  const csv = [header, ...rows].join('\n')
  2294→  const blob = new Blob([csv], { type: 'text/csv;charset=utf-8;' })
  2295→  const url = URL.createObjectURL(blob)
  2296→  const a = document.createElement('a')
  2297→  a.href = url; a.download = `${filename}.csv`; a.click()
  2298→  URL.revokeObjectURL(url)
  2299→  toast.success(`Exported ${data.length} rows as CSV`)
  2300→}
