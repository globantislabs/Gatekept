    16→    16→export async function initDataService(): Promise<boolean> {
    17→    17→  if (shouldUseMockSupabase()) {
    18→    18→    _useReal = false
    19→    19→    return false

    22→    22→  _useReal = connected

    27→    27→  return _useReal

    30→    30→// Helper: try real Supabase, fall back to mock on error
    31→    31→async function tryRealOrMock<T>(realFn: () => Promise<T>, mockFn: () => T): Promise<T> {
    32→    32→  if (!_useReal) return mockFn()
    33→    33→  try {
    34→    34→    return await realFn()
    35→    35→  } catch (err) {
    36→    36→    console.warn('[DataService] Real Supabase query failed, falling back to mock:', err)
    37→    37→    return mockFn()
    38→    38→  }
    39→    39→}

   277→   277→  async getCampaignSales() {
   278→   278→    return tryRealOrMock(
   279→   279→      async () => {
   280→   280→        // For real Supabase: join qr_scans -> users -> orders
   281→   281→        const { data: scans } = await supabase.from('qr_scans').select('campaign_id, user_id')
   282→   282→        const { data: orders } = await supabase.from('orders').select('user_id, amount, status')
   283→   283→        const salesByCampaign: Record<string, { orderCount: number; revenue: number }> = {}
   284→   284→        if (scans && orders) {
   285→   285→          // Build user -> latest campaign mapping from scans
   286→   286→          const userCampaignMap: Record<string, string> = {}
   287→   287→          scans.forEach((s: any) => {
   288→   288→            if (s.user_id && s.campaign_id) userCampaignMap[s.user_id] = s.campaign_id
   289→   289→          })
   290→   290→          // Attribute orders to campaigns
   291→   291→          orders.forEach((o: any) => {
   292→   292→            const campId = userCampaignMap[o.user_id]
   293→   293→            if (campId && ['CONFIRMED', 'SHIPPED', 'DELIVERED'].includes(o.status)) {
   294→   294→              if (!salesByCampaign[campId]) salesByCampaign[campId] = { orderCount: 0, revenue: 0 }
   295→   295→              salesByCampaign[campId].orderCount++
   296→   296→              salesByCampaign[campId].revenue += o.amount || 0
   297→   297→            }
   298→   298→          })
   299→   299→        }
   300→   300→        return { data: salesByCampaign, error: null }
   301→   301→      },
   302→   302→      () => {
   303→   303→        const store = getMockStore()
   304→   304→        const scans = store.getAll<QrScan>('qr_scans')
   305→   305→        const orders = store.getAll<Order>('orders')
   306→   306→        const salesByCampaign: Record<string, { orderCount: number; revenue: number; users: string[] }> = {}
   307→   307→
   308→   308→        // Build user -> campaign mapping from scans (most recent scan wins)
   309→   309→        const userScans = scans.filter(s => s.user_id && s.campaign_id)
   310→   310→        const userCampaignMap: Record<string, string> = {}
   311→   311→        userScans.forEach(s => {
   312→   312→          userCampaignMap[s.user_id!] = s.campaign_id!
   313→   313→        })
   314→   314→
   315→   315→        // Attribute orders to campaigns
   316→   316→        orders.forEach(o => {
   317→   317→          const campId = userCampaignMap[o.user_id]
   318→   318→          if (campId && ['CONFIRMED', 'SHIPPED', 'DELIVERED'].includes(o.status)) {
   319→   319→            if (!salesByCampaign[campId]) salesByCampaign[campId] = { orderCount: 0, revenue: 0, users: [] }
   320→   320→            salesByCampaign[campId].orderCount++
   321→   321→            salesByCampaign[campId].revenue += o.amount || 0
   322→   322→            if (!salesByCampaign[campId].users.includes(o.user_id)) {
   323→   323→              salesByCampaign[campId].users.push(o.user_id)
   324→   324→            }
   325→   325→          }
   326→   326→        })
   327→   327→
   328→   328→        return { data: salesByCampaign, error: null }
   329→   329→      }
   330→   330→    )
   331→   331→  },

   583→   583→          supabase.from('users_profile').select('*', { count: 'exact', head: true }),
   584→   584→          supabase.from('qr_scans').select('*', { count: 'exact', head: true }),

   586→   586→          supabase.from('campaigns').select('channel, status'),
   587→   587→        ])

   605→   605→        const scansByCampaign: Record<string, number> = {}
   606→   606→        scans.forEach(s => { if (s.campaign_id) scansByCampaign[s.campaign_id] = (scansByCampaign[s.campaign_id] || 0) + 1 })

   621→   621→
   622→   622→        return {
