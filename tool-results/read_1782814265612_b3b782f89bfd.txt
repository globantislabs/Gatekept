   786→  // ============================================
   787→  // CRUD Operations (mirror Supabase from().select() API)
   788→  // ============================================
   789→
   790→  getAll<T>(table: TableName): T[] {
   791→    return this.data[table] as T[]
   792→  }
   793→
   794→  getById<T>(table: TableName, id: string): T | undefined {
   795→    return (this.data[table] as any[]).find((item: any) => item.id === id) as T | undefined
   796→  }
   797→
   798→  find<T>(table: TableName, filter: Partial<T>): T[] {
   799→    return (this.data[table] as any[]).filter((item: any) => {
   800→      return Object.entries(filter).every(([key, value]) => {
   801→        if (value === undefined || value === null) return true
   802→        return item[key] === value
   803→      })
   804→    }) as T[]
   805→  }
   806→
   807→  findOne<T>(table: TableName, filter: Partial<T>): T | undefined {
   808→    return this.find<T>(table, filter)[0]
   809→  }
   810→
   811→  insert<T>(table: TableName, record: any): T {
   812→    const item = { ...record, id: record.id || `${table.slice(0, 4)}_${Date.now()}_${Math.random().toString(36).slice(2, 6)}` }
   813→    ;(this.data[table] as any[]).push(item)
   814→    this.save()
   815→    return item as T
   816→  }
   817→
   818→  update<T>(table: TableName, id: string, updates: Partial<T>): T | undefined {
   819→    const idx = (this.data[table] as any[]).findIndex((item: any) => item.id === id)
   820→    if (idx === -1) return undefined
   821→    this.data[table][idx] = { ...this.data[table][idx], ...updates }
   822→    this.save()
   823→    return this.data[table][idx] as T
   824→  }
   825→
   826→  delete(table: TableName, id: string): boolean {
   827→    const idx = (this.data[table] as any[]).findIndex((item: any) => item.id === id)
   828→    if (idx === -1) return false
   829→    this.data[table].splice(idx, 1)
   830→    this.save()
   831→    return true
   832→  }
   833→
   834→  count(table: TableName, filter?: Record<string, any>): number {
   835→    if (filter) return this.find(table, filter).length
   836→    return this.data[table].length
   837→  }

   841→    this.seed()
   842→  }
