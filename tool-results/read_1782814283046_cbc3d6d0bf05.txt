   105→   105→function AnimatedSection({ children, className = '', id }: { children: React.ReactNode; className?: string; id?: string }) {
   106→   106→  const ref = useRef(null)
   107→   107→  const isInView = useInView(ref, { once: true, margin: '-80px' })
   108→   108→  return (
   109→   109→    <motion.section
   110→   110→      ref={ref}
   111→   111→      id={id}
   112→   112→      initial="hidden"
   113→   113→      animate={isInView ? 'visible' : 'hidden'}
   114→   114→      variants={staggerContainer}
   115→   115→      className={className}
   116→   116→    >
   117→   117→      {children}
   118→   118→    </motion.section>
   119→   119→  )
   120→   120→}

   125→   125→function AnimatedCounter({ end, suffix = '', prefix = '' }: { end: number; suffix?: string; prefix?: string }) {
   126→   126→  const [count, setCount] = useState(0)
   127→   127→  const ref = useRef(null)
   128→   128→  const isInView = useInView(ref, { once: true })
   129→   129→  useEffect(() => {
   130→   130→    if (!isInView) return
   131→   131→    let start = 0
   132→   132→    const duration = 2000
   133→   133→    const increment = end / (duration / 16)
   134→   134→    const timer = setInterval(() => {
   135→   135→      start += increment
   136→   136→      if (start >= end) { setCount(end); clearInterval(timer) }
   137→   137→      else setCount(Math.floor(start))
   138→   138→    }, 16)
   139→   139→    return () => clearInterval(timer)
   140→   140→  }, [isInView, end])
   141→   141→  return <span ref={ref}>{prefix}{count.toLocaleString()}{suffix}</span>
   142→   142→}
