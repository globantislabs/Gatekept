   110→      ref={ref}
   111→      id={id}
   112→      initial="hidden"
   113→      animate={isInView ? 'visible' : 'hidden'}
   114→      variants={staggerContainer}
   115→      className={className}
   116→    >
   117→      {children}
   118→    </motion.section>
   119→  )

   123→// COUNTER ANIMATION COMPONENT
   124→// ============================================================
   125→function AnimatedCounter({ end, suffix = '', prefix = '' }: { end: number; suffix?: string; prefix?: string }) {
   126→  const [count, setCount] = useState(0)
   127→  const ref = useRef(null)
   128→  const isInView = useInView(ref, { once: true })
   129→  useEffect(() => {
   130→    if (!isInView) return
   131→    let start = 0
   132→    const duration = 2000
   133→    const increment = end / (duration / 16)
   134→    const timer = setInterval(() => {
   135→      start += increment
   136→      if (start >= end) { setCount(end); clearInterval(timer) }
   137→      else setCount(Math.floor(start))
   138→    }, 16)
   139→    return () => clearInterval(timer)
   140→  }, [isInView, end])
   141→  return <span ref={ref}>{prefix}{count.toLocaleString()}{suffix}</span>
   142→}
