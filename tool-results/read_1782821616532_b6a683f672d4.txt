    13→// Whether to use real Supabase (only if configured AND tables exist)
    14→let _useReal = false
    15→
    16→export async function initDataService(): Promise<boolean> {
    17→  if (shouldUseMockSupabase()) {
    18→    _useReal = false
    19→    return false
    20→  }
    21→  const connected = await testSupabaseConnection()
    22→  _useReal = connected
    23→  return connected
    24→}
    25→
    26→export function isUsingRealSupabase(): boolean {
    27→  return _useReal
    28→}
    29→
    30→// Helper: try real Supabase, fall back to mock on error
    31→async function tryRealOrMock<T>(realFn: () => Promise<T>, mockFn: () => T): Promise<T> {
    32→  if (!_useReal) return mockFn()
    33→  try {
    34→    return await realFn()
    35→  } catch (err) {
    36→    console.warn('[DataService] Real Supabase query failed, falling back to mock:', err)
    37→    return mockFn()
    38→  }
    39→}

   298→  async getCampaignSales() {
   299→    return tryRealOrMock(
   300→      async () => {
   301→        // For real Supabase: join qr_scans -> users -> orders
   302→        const { data: scans } = await supabase.from('qr_scans').select('campaign_id, user_id')
   303→        const { data: orders } = await supabase.from('orders').select('user_id, amount, status')
   304→        const salesByCampaign: Record<string, { orderCount: number; revenue: number }> = {}
   305→        if (scans && orders) {
   306→          // Build user -> latest campaign mapping from scans
   307→          const userCampaignMap: Record<string, string> = {}
   308→          scans.forEach((s: any) => {
   309→            if (s.user_id && s.campaign_id) userCampaignMap[s.user_id] = s.campaign_id
   310→          })
   311→          // Attribute orders to campaigns
   312→          orders.forEach((o: any) => {
   313→            const campId = userCampaignMap[o.user_id]
   314→            if (campId && ['CONFIRMED', 'SHIPPED', 'DELIVERED'].includes(o.status)) {
   315→              if (!salesByCampaign[campId]) salesByCampaign[campId] = { orderCount: 0, revenue: 0 }
   316→              salesByCampaign[campId].orderCount++
   317→              salesByCampaign[campId].revenue += o.amount || 0
   318→            }
   319→          })
   320→        }
   321→        return { data: salesByCampaign, error: null }
   322→      },
   323→      () => {
   324→        const store = getMockStore()
   325→        const scans = store.getAll<QrScan>('qr_scans')
   326→        const orders = store.getAll<Order>('orders')
   327→        const salesByCampaign: Record<string, { orderCount: number; revenue: number; users: string[] }> = {}
   328→
   329→        // Build user -> campaign mapping from scans (most recent scan wins)
   330→        const userScans = scans.filter(s => s.user_id && s.campaign_id)
   331→        const userCampaignMap: Record<string, string> = {}
   332→        userScans.forEach(s => {
   333→          userCampaignMap[s.user_id!] = s.campaign_id!
   334→        })
   335→
   336→        // Attribute orders to campaigns
   337→        orders.forEach(o => {
   338→          const campId = userCampaignMap[o.user_id]
   339→          if (campId && ['CONFIRMED', 'SHIPPED', 'DELIVERED'].includes(o.status)) {
   340→            if (!salesByCampaign[campId]) salesByCampaign[campId] = { orderCount: 0, revenue: 0, users: [] }
   341→            salesByCampaign[campId].orderCount++
   342→            salesByCampaign[campId].revenue += o.amount || 0
   343→            if (!salesByCampaign[campId].users.includes(o.user_id)) {
   344→              salesByCampaign[campId].users.push(o.user_id)
   345→            }
   346→          }
   347→        })
   348→
   349→        return { data: salesByCampaign, error: null }
   350→      }
   351→    )
   352→  },
   353→}

   534→          shipping_address: shippingAddress, created_at: now, updated_at: now,
   535→        })
   536→        for (const oi of orderItems) {
   537→          store.insert('order_items', { order_id: order.id, ...oi })

   599→export const adminStatsService = {
   600→  async getDashboardStats() {
   601→    return tryRealOrMock(
   602→      async () => {
   603→        const [users, scans, orders, campaigns] = await Promise.all([
   604→          supabase.from('users_profile').select('*', { count: 'exact', head: true }),
   605→          supabase.from('qr_scans').select('*', { count: 'exact', head: true }),
   606→          supabase.from('orders').select('status, amount'),
   607→          supabase.from('campaigns').select('channel, status'),
   608→        ])
   609→        return { data: { totalUsers: users.count, totalScans: scans.count, totalOrders: orders.data?.length, totalCampaigns: campaigns.data?.length }, error: null }
   610→      },
   611→      () => {
   612→        const store = getMockStore()
   613→        const totalUsers = store.count('users_profile')
   614→        const totalScans = store.count('qr_scans')
   615→        const totalOrders = store.count('orders')
   616→        const learningCompleted = store.count('users_profile', { learning_completed: true } as any)
   617→        const totalCampaigns = store.count('campaigns')
   618→        const orders = store.getAll<Order>('orders')
   619→        const totalRevenue = orders.filter(o => ['CONFIRMED', 'SHIPPED', 'DELIVERED'].includes(o.status)).reduce((sum, o) => sum + o.amount, 0)
   620→        const conversionRate = totalUsers > 0 ? ((learningCompleted / totalUsers) * 100).toFixed(1) : '0'
   621→
   622→        const ordersByStatus: Record<string, number> = {}
   623→        orders.forEach(o => { ordersByStatus[o.status] = (ordersByStatus[o.status] || 0) + 1 })
   624→
   625→        const scans = store.getAll<QrScan>('qr_scans')
   626→        const scansByCampaign: Record<string, number> = {}
   627→        scans.forEach(s => { if (s.campaign_id) scansByCampaign[s.campaign_id] = (scansByCampaign[s.campaign_id] || 0) + 1 })
   628→
   629→        const campaigns = store.getAll<Campaign>('campaigns')
   630→        const campaignsByChannel: Record<string, number> = {}
   631→        campaigns.forEach(c => { campaignsByChannel[c.channel] = (campaignsByChannel[c.channel] || 0) + 1 })
   632→
   633→        const recentUsers = [...store.getAll<UserProfile>('users_profile')].sort((a, b) => new Date(b.created_at).getTime() - new Date(a.created_at).getTime()).slice(0, 5)
   634→
   635→        const recentOrders = [...orders].sort((a, b) => new Date(b.created_at).getTime() - new Date(a.created_at).getTime()).slice(0, 5).map(o => ({
   636→          ...o,
   637→          user: store.findOne<UserProfile>('users_profile', { user_id: o.user_id } as any),
   638→          items: store.find<OrderItem>('order_items', { order_id: o.id } as any).map(oi => ({
   639→            ...oi, product: store.getById<Product>('products', oi.product_id),
   640→          })),
   641→        }))
   642→
   643→        return {
   644→          data: {
   645→            totalUsers, totalScans, totalOrders, learningCompleted, totalCampaigns, totalRevenue,
   646→            conversionRate, ordersByStatus, scansByCampaign, campaignsByChannel, recentUsers, recentOrders,
   647→          }, error: null,
   648→        }
   649→      }
   650→    )
   651→  },
   652→}

   789→        return { data: getMockStore().insert('product_learning_progress', { user_id: userId, product_id: productId, video_progress: videoProgress, quiz_answers: {}, quiz_completed: false, quiz_score: 0, status: 'IN_PROGRESS', created_at: now, updated_at: now }), error: null }
   790→      }
   791→    )
   792→  },
   793→  async submitQuiz(userId: string, productId: string, quizAnswers: Record<string, number>, score: number, allCorrect: boolean) {
   794→    return tryRealOrMock(
   795→      async () => {
   796→        const status = allCorrect ? 'UNLOCKED' : 'PENDING_REEVALUATION'
   797→        const { data, error } = await supabase.from('product_learning_progress').update({
   798→          quiz_answers: quizAnswers, quiz_completed: allCorrect, quiz_score: score, status, completed_at: allCorrect ? new Date().toISOString() : null,
   799→        }).eq('user_id', userId).eq('product_id', productId).select().single()
   800→        return { data, error }
   801→      },
   802→      () => {
   803→        const existing = getMockStore().findOne<ProductLearningProgress>('product_learning_progress', { user_id: userId, product_id: productId } as any)
   804→        const now = new Date().toISOString()
   805→        const status = allCorrect ? 'UNLOCKED' : 'PENDING_REEVALUATION'
   806→        if (existing) {
   807→          // Smart remediation: if not all correct, mark only incorrect video modules as incomplete
   808→          if (!allCorrect && existing.video_progress) {
   809→            // Get quizzes for this product to find which videos need review
   810→            const quizzes = getMockStore().find<ProductQuiz>('product_quizzes', { product_id: productId, active: true } as any)
   811→            const updatedVideoProgress = { ...existing.video_progress }
   812→            for (const quiz of quizzes) {
   813→              const userAnswer = quizAnswers[quiz.id]
   814→              if (userAnswer !== undefined && userAnswer !== quiz.answer) {
   815→                // Mark this quiz's linked video as incomplete (0%)
   816→                updatedVideoProgress[quiz.video_id] = 0
   817→              }
   818→            }
   819→            return { data: getMockStore().update('product_learning_progress', existing.id, {
   820→              quiz_answers: quizAnswers, quiz_completed: false, quiz_score: score, status, video_progress: updatedVideoProgress, updated_at: now,
   821→            }), error: null }
   822→          }
   823→          return { data: getMockStore().update('product_learning_progress', existing.id, {
   824→            quiz_answers: quizAnswers, quiz_completed: allCorrect, quiz_score: score, status, completed_at: allCorrect ? now : undefined, updated_at: now,
   825→          }), error: null }
   826→        }
   827→        return { data: getMockStore().insert('product_learning_progress', {
   828→          user_id: userId, product_id: productId, video_progress: {}, quiz_answers: quizAnswers, quiz_completed: allCorrect, quiz_score: score, status, completed_at: allCorrect ? now : undefined, created_at: now, updated_at: now,
   829→        }), error: null }
   830→      }
   831→    )
   832→  },
