    13→// Whether to use real Supabase (only if configured AND tables exist)
    14→let _useReal = false
    15→
    16→export async function initDataService(): Promise<boolean> {
    17→  if (shouldUseMockSupabase()) {
    18→    _useReal = false
    19→    return false
    20→  }
    21→  const connected = await testSupabaseConnection()
    22→  _useReal = connected
    23→  return connected
    24→}
    25→
    26→export function isUsingRealSupabase(): boolean {
    27→  return _useReal
    28→}
    29→
    30→// Helper: try real Supabase, fall back to mock on error
    31→async function tryRealOrMock<T>(realFn: () => Promise<T>, mockFn: () => T): Promise<T> {
    32→  if (!_useReal) return mockFn()
    33→  try {
    34→    return await realFn()
    35→  } catch (err) {
    36→    console.warn('[DataService] Real Supabase query failed, falling back to mock:', err)
    37→    return mockFn()
    38→  }
    39→}

   298→  async getCampaignSales() {
   299→    return tryRealOrMock(
   300→      async () => {
   301→        // For real Supabase: join qr_scans -> users -> orders
   302→        const { data: scans } = await supabase.from('qr_scans').select('campaign_id, user_id')
   303→        const { data: orders } = await supabase.from('orders').select('user_id, amount, status')
   304→        const salesByCampaign: Record<string, { orderCount: number; revenue: number }> = {}
   305→        if (scans && orders) {
   306→          // Build user -> latest campaign mapping from scans
   307→          const userCampaignMap: Record<string, string> = {}
   308→          scans.forEach((s: any) => {
   309→            if (s.user_id && s.campaign_id) userCampaignMap[s.user_id] = s.campaign_id
   310→          })
   311→          // Attribute orders to campaigns
   312→          orders.forEach((o: any) => {
   313→            const campId = userCampaignMap[o.user_id]
   314→            if (campId && ['CONFIRMED', 'SHIPPED', 'DELIVERED'].includes(o.status)) {
   315→              if (!salesByCampaign[campId]) salesByCampaign[campId] = { orderCount: 0, revenue: 0 }
   316→              salesByCampaign[campId].orderCount++
   317→              salesByCampaign[campId].revenue += o.amount || 0
   318→            }
   319→          })
   320→        }
   321→        return { data: salesByCampaign, error: null }
   322→      },
   323→      () => {
   324→        const store = getMockStore()
   325→        const scans = store.getAll<QrScan>('qr_scans')
   326→        const orders = store.getAll<Order>('orders')
   327→        const salesByCampaign: Record<string, { orderCount: number; revenue: number; users: string[] }> = {}
   328→
   329→        // Build user -> campaign mapping from scans (most recent scan wins)
   330→        const userScans = scans.filter(s => s.user_id && s.campaign_id)
   331→        const userCampaignMap: Record<string, string> = {}
   332→        userScans.forEach(s => {
   333→          userCampaignMap[s.user_id!] = s.campaign_id!
   334→        })
   335→
   336→        // Attribute orders to campaigns
   337→        orders.forEach(o => {
   338→          const campId = userCampaignMap[o.user_id]
   339→          if (campId && ['CONFIRMED', 'SHIPPED', 'DELIVERED'].includes(o.status)) {
   340→            if (!salesByCampaign[campId]) salesByCampaign[campId] = { orderCount: 0, revenue: 0, users: [] }
   341→            salesByCampaign[campId].orderCount++
   342→            salesByCampaign[campId].revenue += o.amount || 0
   343→            if (!salesByCampaign[campId].users.includes(o.user_id)) {
   344→              salesByCampaign[campId].users.push(o.user_id)
   345→            }
   346→          }
   347→        })
   348→
   349→        return { data: salesByCampaign, error: null }
   350→      }
   351→    )
   352→  },
   353→}

   659→  async getDashboardStats() {
   660→    return tryRealOrMock(
   661→      async () => {
   662→        const [users, scans, orders, campaigns] = await Promise.all([
   663→          supabase.from('users_profile').select('*', { count: 'exact', head: true }),
   664→          supabase.from('qr_scans').select('*', { count: 'exact', head: true }),
   665→          supabase.from('orders').select('status, amount'),
   666→          supabase.from('campaigns').select('channel, status'),
   667→        ])
   668→        return { data: { totalUsers: users.count, totalScans: scans.count, totalOrders: orders.data?.length, totalCampaigns: campaigns.data?.length }, error: null }
   669→      },
   670→      () => {
   671→        const store = getMockStore()
   672→        const totalUsers = store.count('users_profile')
   673→        const totalScans = store.count('qr_scans')
   674→        const totalOrders = store.count('orders')
   675→        const learningCompleted = store.count('users_profile', { learning_completed: true } as any)
   676→        const totalCampaigns = store.count('campaigns')
   677→        const orders = store.getAll<Order>('orders')
   678→        const totalRevenue = orders.filter(o => ['CONFIRMED', 'SHIPPED', 'DELIVERED'].includes(o.status)).reduce((sum, o) => sum + o.amount, 0)
   679→        const conversionRate = totalUsers > 0 ? ((learningCompleted / totalUsers) * 100).toFixed(1) : '0'
   680→
   681→        const ordersByStatus: Record<string, number> = {}
   682→        orders.forEach(o => { ordersByStatus[o.status] = (ordersByStatus[o.status] || 0) + 1 })
   683→
   684→        const scans = store.getAll<QrScan>('qr_scans')
   685→        const scansByCampaign: Record<string, number> = {}
   686→        scans.forEach(s => { if (s.campaign_id) scansByCampaign[s.campaign_id] = (scansByCampaign[s.campaign_id] || 0) + 1 })
   687→
   688→        const campaigns = store.getAll<Campaign>('campaigns')
   689→        const campaignsByChannel: Record<string, number> = {}
   690→        campaigns.forEach(c => { campaignsByChannel[c.channel] = (campaignsByChannel[c.channel] || 0) + 1 })
   691→
   692→        const recentUsers = [...store.getAll<UserProfile>('users_profile')].sort((a, b) => new Date(b.created_at).getTime() - new Date(a.created_at).getTime()).slice(0, 5)
   693→
   694→        const recentOrders = [...orders].sort((a, b) => new Date(b.created_at).getTime() - new Date(a.created_at).getTime()).slice(0, 5).map(o => ({
   695→          ...o,
   696→          user: store.findOne<UserProfile>('users_profile', { user_id: o.user_id } as any),
   697→          items: store.find<OrderItem>('order_items', { order_id: o.id } as any).map(oi => ({
   698→            ...oi, product: store.getById<Product>('products', oi.product_id),
   699→          })),
   700→        }))
   701→
   702→        return {
   703→          data: {
   704→            totalUsers, totalScans, totalOrders, learningCompleted, totalCampaigns, totalRevenue,
   705→            conversionRate, ordersByStatus, scansByCampaign, campaignsByChannel, recentUsers, recentOrders,
   706→          }, error: null,
   707→        }
   708→      }
   709→    )
   710→  },
