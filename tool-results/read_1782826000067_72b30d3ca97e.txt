   796→  // ============================================
   797→  // CRUD Operations (mirror Supabase from().select() API)
   798→  // ============================================
   799→
   800→  getAll<T>(table: TableName): T[] {
   801→    return this.data[table] as T[]
   802→  }
   803→
   804→  getById<T>(table: TableName, id: string): T | undefined {
   805→    return (this.data[table] as any[]).find((item: any) => item.id === id) as T | undefined
   806→  }
   807→
   808→  find<T>(table: TableName, filter: Partial<T>): T[] {
   809→    return (this.data[table] as any[]).filter((item: any) => {
   810→      return Object.entries(filter).every(([key, value]) => {
   811→        if (value === undefined || value === null) return true
   812→        return item[key] === value
   813→      })
   814→    }) as T[]
   815→  }
   816→
   817→  findOne<T>(table: TableName, filter: Partial<T>): T | undefined {
   818→    return this.find<T>(table, filter)[0]
   819→  }
   820→
   821→  insert<T>(table: TableName, record: any): T {
   822→    const item = { ...record, id: record.id || `${table.slice(0, 4)}_${Date.now()}_${Math.random().toString(36).slice(2, 6)}` }
   823→    ;(this.data[table] as any[]).push(item)
   824→    this.save()
   825→    return item as T
   826→  }
   827→
   828→  update<T>(table: TableName, id: string, updates: Partial<T>): T | undefined {
   829→    const idx = (this.data[table] as any[]).findIndex((item: any) => item.id === id)
   830→    if (idx === -1) return undefined
   831→    this.data[table][idx] = { ...this.data[table][idx], ...updates }
   832→    this.save()
   833→    return this.data[table][idx] as T
   834→  }
   835→
   836→  delete(table: TableName, id: string): boolean {
   837→    const idx = (this.data[table] as any[]).findIndex((item: any) => item.id === id)
   838→    if (idx === -1) return false
   839→    this.data[table].splice(idx, 1)
   840→    this.save()
   841→    return true
   842→  }
   843→
   844→  count(table: TableName, filter?: Record<string, any>): number {
   845→    if (filter) return this.find(table, filter).length
   846→    return this.data[table].length
   847→  }

   850→  reset() {
   851→    this.seed()
   852→  }
