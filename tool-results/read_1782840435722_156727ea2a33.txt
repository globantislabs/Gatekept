   113→      initial="hidden"
   114→      animate={isInView ? 'visible' : 'hidden'}
   115→      variants={staggerContainer}
   116→      className={className}
   117→    >
   118→      {children}

   120→  )

   126→function AnimatedCounter({ end, suffix = '', prefix = '' }: { end: number; suffix?: string; prefix?: string }) {
   127→  const [count, setCount] = useState(0)
   128→  const ref = useRef(null)
   129→  const isInView = useInView(ref, { once: true })
   130→  useEffect(() => {
   131→    if (!isInView) return
   132→    let start = 0
   133→    const duration = 2000
   134→    const increment = end / (duration / 16)
   135→    const timer = setInterval(() => {
   136→      start += increment
   137→      if (start >= end) { setCount(end); clearInterval(timer) }
   138→      else setCount(Math.floor(start))
   139→    }, 16)
   140→    return () => clearInterval(timer)
   141→  }, [isInView, end])
   142→  return <span ref={ref}>{prefix}{count.toLocaleString()}{suffix}</span>
   143→}
