    37→export function ProductLearningModule() {
    38→  const { selectedProductId, user, navigateTo, currentVideoIndex, setCurrentVideoIndex, setCurrentQuizIndex } = useAppStore()
    39→
    40→  const [product, setProduct] = useState<Product | null>(null)
    41→  const [videos, setVideos] = useState<ProductVideo[]>([])
    42→  const [progress, setProgress] = useState<ProductLearningProgress | null>(null)
    43→  const [loading, setLoading] = useState(true)
    44→  const [playing, setPlaying] = useState(false)
    45→  const [videoProgress, setVideoProgress] = useState(0) // 0-100 for current video
    46→  const progressIntervalRef = useRef<ReturnType<typeof setInterval> | null>(null)
    47→
    48→  // ─── Fetch data ────────────────────────────────────────
    49→  useEffect(() => {
    50→    if (!selectedProductId) return
    51→    let cancelled = false
    52→    async function load() {
    53→      setLoading(true)
    54→      try {
    55→        const [prodRes, vidRes, progRes] = await Promise.all([
    56→          productService.getById(selectedProductId),
    57→          productVideoService.getByProduct(selectedProductId),
    58→          user ? productLearningService.getProgress(user.user_id, selectedProductId) : Promise.resolve({ data: null, error: null }),
    59→        ])
    60→        if (cancelled) return
    61→        setProduct(prodRes.data || null)
    62→        setVideos(vidRes.data || [])
    63→        setProgress(progRes.data || null)
    64→      } catch (err) {
    65→        console.error('Failed to load learning data:', err)
    66→      } finally {
    67→        if (!cancelled) setLoading(false)
    68→      }
    69→    }
    70→    load()
    71→    return () => { cancelled = true }
    72→  }, [selectedProductId, user])
    73→
    74→  // ─── Video progress for current video from learning progress ──
    75→  const currentVideo = videos[currentVideoIndex] || null
    76→  const currentVideoSavedProgress = currentVideo && progress?.video_progress
    77→    ? (progress.video_progress[currentVideo.id] ?? 0)
    78→    : 0
    79→
    80→  // ─── Reset local state when switching videos ───────────
    81→  useEffect(() => {
    82→    setPlaying(false)
    83→    setVideoProgress(currentVideoSavedProgress)
    84→    if (progressIntervalRef.current) {
    85→      clearInterval(progressIntervalRef.current)
    86→      progressIntervalRef.current = null
    87→    }
    88→  }, [currentVideoIndex, currentVideo?.id])
    89→
    90→  // ─── Simulated video playback ──────────────────────────
    91→  useEffect(() => {
    92→    if (playing && videoProgress < 100) {
    93→      progressIntervalRef.current = setInterval(() => {
    94→        setVideoProgress(prev => {
    95→          const next = Math.min(prev + 2, 100)
    96→          if (next >= 100) {
    97→            setPlaying(false)
    98→            saveProgress(next)
    99→          }
   100→          return next
   101→        })
   102→      }, 150)
   103→    }
   104→    return () => {
   105→      if (progressIntervalRef.current) {
   106→        clearInterval(progressIntervalRef.current)
   107→        progressIntervalRef.current = null
   108→      }
   109→    }
   110→  }, [playing])
   111→
   112→  // ─── Save progress to backend ──────────────────────────
   113→  const saveProgress = async (pct: number) => {
   114→    if (!user || !selectedProductId || !currentVideo) return
   115→    const updatedVP = { ...(progress?.video_progress || {}), [currentVideo.id]: pct }
   116→    const res = await productLearningService.updateVideoProgress(user.user_id, selectedProductId, updatedVP)
   117→    if (res.data) setProgress(res.data)
   118→  }
   119→
   120→  const handlePause = () => {
   121→    setPlaying(false)
   122→    saveProgress(videoProgress)
   123→  }
   124→
   125→  const handleSeek = (e: React.MouseEvent<HTMLDivElement>) => {
   126→    const rect = e.currentTarget.getBoundingClientRect()
   127→    const pct = Math.round(((e.clientX - rect.left) / rect.width) * 100)
   128→    setVideoProgress(pct)
   129→    saveProgress(pct)
   130→  }
   131→
   132→  // ─── Overall progress ──────────────────────────────────
   133→  const overallProgress = videos.length > 0
   134→    ? Math.round(videos.reduce((sum, v) => sum + (progress?.video_progress?.[v.id] ?? 0), 0) / videos.length)
   135→    : 0
   136→
   137→  const allVideosComplete = videos.length > 0 && videos.every(v => (progress?.video_progress?.[v.id] ?? 0) >= 100)
   138→
   139→  // ─── Loading state ─────────────────────────────────────
   140→  if (loading) {
   141→    return (
   142→      <div className="flex items-center justify-center min-h-[60vh]">
   143→        <div className="flex flex-col items-center gap-3">
   144→          <RefreshCw className="h-8 w-8 animate-spin" style={{ color: BRAND.green }} />
   145→          <p className="text-sm" style={{ color: BRAND.muted }}>Loading learning content...</p>
   146→        </div>
   147→      </div>
   148→    )
   149→  }
   150→
   151→  if (!product || videos.length === 0) {
   152→    return (
   153→      <div className="flex items-center justify-center min-h-[60vh]">
   154→        <Card className="max-w-md w-full" style={{ borderColor: BRAND.surface }}>
   155→          <CardContent className="p-6 text-center">
   156→            <AlertCircle className="h-10 w-10 mx-auto mb-3" style={{ color: BRAND.muted }} />
   157→            <p className="font-heading text-lg" style={{ color: BRAND.dark }}>No learning content available</p>
   158→            <p className="text-sm mt-1" style={{ color: BRAND.muted }}>This product doesn't have learning videos yet.</p>
   159→            <Button
   160→              className="mt-4"
   161→              style={{ backgroundColor: BRAND.green }}
   162→              onClick={() => navigateTo('products')}
   163→            >
   164→              <ArrowLeft className="h-4 w-4 mr-2" /> Back to Products
   165→            </Button>
   166→          </CardContent>
   167→        </Card>
   168→      </div>
   169→    )
   170→  }
   171→
   172→  return (
   173→    <motion.div {...fadeInUp} transition={{ duration: 0.4 }} className="max-w-6xl mx-auto px-4 py-6">
   174→      {/* ─── Header ────────────────────────────────────── */}
   175→      <div className="flex items-center gap-3 mb-6">
   176→        <Button
   177→          variant="ghost"
   178→          size="sm"
   179→          onClick={() => navigateTo('product-detail')}
   180→          className="p-2"
   181→        >
   182→          <ArrowLeft className="h-5 w-5" style={{ color: BRAND.dark }} />
   183→        </Button>
   184→        <div className="flex-1">
   185→          <h1 className="font-heading text-xl sm:text-2xl font-bold" style={{ color: BRAND.dark }}>
   186→            Learn: {product.name}
   187→          </h1>
   188→          <p className="text-sm" style={{ color: BRAND.muted }}>
   189→            Complete all videos to unlock the quiz
   190→          </p>
   191→        </div>
   192→        <Badge
   193→          style={{
   194→            backgroundColor: allVideosComplete ? BRAND.lime : BRAND.surface,
   195→            color: allVideosComplete ? BRAND.dark : BRAND.muted,

   197→          className="text-xs font-medium"
   198→        >
   199→          {overallProgress}% Complete
   200→        </Badge>

   247→                  whileTap={{ scale: 0.95 }}
   248→                  className="z-10"
   249→                >
   250→                  {videoProgress >= 100 ? (
   251→                    <div className="flex flex-col items-center gap-2">
   252→                      <CheckCircle className="h-16 w-16" style={{ color: BRAND.lime }} />
   253→                      <span className="text-sm font-medium" style={{ color: BRAND.lime }}>Completed</span>
   254→                    </div>
   255→                  ) : (

   257→                      className="h-16 w-16 rounded-full flex items-center justify-center backdrop-blur-sm"

   328→                        onClick={() => setCurrentVideoIndex(currentVideoIndex + 1)}

   331→                        Next <ArrowRight className="h-4 w-4 ml-1" />

   334→                  </div>
   335→                </div>
   336→              </CardContent>
   337→            </Card>
   338→          </motion.div>
   339→
   340→          {/* ─── Take Quiz CTA ─────────────────────────── */}

   344→                className="border-2"
   345→                style={{ borderColor: BRAND.lime, backgroundColor: `${BRAND.lime}15` }}

   669→              <ArrowLeft className="h-4 w-4 mr-2" /> Back to Learning

   677→  // ─── Results Screen ────────────────────────────────────

   679→    return (

   683→          variant="ghost"
   684→          size="sm"
   685→          onClick={() => navigateTo('product-learning')}
   686→          className="mb-4 p-2"
   687→        >
   688→          <ArrowLeft className="h-5 w-5 mr-1" style={{ color: BRAND.dark }} />

   695→            className="border-2"
   696→            style={{
   697→              borderColor: allCorrect ? BRAND.lime : BRAND.green,

   699→          >
   700→            <CardHeader className="text-center pb-2">
   701→              <motion.div
   702→                initial={{ scale: 0 }}

   704→                transition={{ type: 'spring', stiffness: 200, delay: 0.2 }}
   705→              >
   706→                {allCorrect ? (
   707→                  <CheckCircle className="h-16 w-16 mx-auto" style={{ color: BRAND.lime }} />
   708→                ) : (
   709→                  <AlertCircle className="h-16 w-16 mx-auto" style={{ color: BRAND.green }} />
   710→                )}
   711→              </motion.div>
   712→              <CardTitle className="font-heading text-2xl mt-3" style={{ color: BRAND.dark }}>
   713→                {allCorrect ? 'Congratulations!' : 'Almost There!'}
   714→              </CardTitle>
   715→              <CardDescription style={{ color: BRAND.muted }}>
   716→                {allCorrect
   717→                  ? `You've unlocked ${product.name}!`
   718→                  : 'Review the highlighted videos and try again'}
   719→              </CardDescription>
   720→            </CardHeader>
   721→
   722→            <CardContent className="space-y-4">
   723→              {/* Score */}
   724→              <div
   725→                className="text-center py-4 rounded-lg"

   801→            </CardContent>

   803→            <CardFooter className="flex flex-col sm:flex-row gap-3 p-6 pt-2">

   805→                <Button
   806→                  className="flex-1 font-heading"
   807→                  style={{ backgroundColor: BRAND.blue, color: '#fff' }}

   844→          variant="ghost"
   845→          size="sm"
   846→          onClick={() => navigateTo('product-learning')}
   847→          className="p-2"
   848→        >
   849→          <ArrowLeft className="h-5 w-5" style={{ color: BRAND.dark }} />

   852→          <h1 className="font-heading text-xl sm:text-2xl font-bold" style={{ color: BRAND.dark }}>
   853→            Quiz: {product.name}

   855→          <p className="text-sm" style={{ color: BRAND.muted }}>

  1005→
