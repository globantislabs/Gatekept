   792→  // ============================================
   793→  // CRUD Operations (mirror Supabase from().select() API)
   794→  // ============================================
   795→
   796→  getAll<T>(table: TableName): T[] {
   797→    return this.data[table] as T[]
   798→  }
   799→
   800→  getById<T>(table: TableName, id: string): T | undefined {
   801→    return (this.data[table] as any[]).find((item: any) => item.id === id) as T | undefined
   802→  }
   803→
   804→  find<T>(table: TableName, filter: Partial<T>): T[] {
   805→    return (this.data[table] as any[]).filter((item: any) => {
   806→      return Object.entries(filter).every(([key, value]) => {
   807→        if (value === undefined || value === null) return true
   808→        return item[key] === value
   809→      })
   810→    }) as T[]
   811→  }
   812→
   813→  findOne<T>(table: TableName, filter: Partial<T>): T | undefined {
   814→    return this.find<T>(table, filter)[0]
   815→  }
   816→
   817→  insert<T>(table: TableName, record: any): T {
   818→    const item = { ...record, id: record.id || `${table.slice(0, 4)}_${Date.now()}_${Math.random().toString(36).slice(2, 6)}` }
   819→    ;(this.data[table] as any[]).push(item)
   820→    this.save()
   821→    return item as T
   822→  }
   823→
   824→  update<T>(table: TableName, id: string, updates: Partial<T>): T | undefined {
   825→    const idx = (this.data[table] as any[]).findIndex((item: any) => item.id === id)
   826→    if (idx === -1) return undefined
   827→    this.data[table][idx] = { ...this.data[table][idx], ...updates }
   828→    this.save()
   829→    return this.data[table][idx] as T
   830→  }
   831→
   832→  delete(table: TableName, id: string): boolean {
   833→    const idx = (this.data[table] as any[]).findIndex((item: any) => item.id === id)
   834→    if (idx === -1) return false
   835→    this.data[table].splice(idx, 1)
   836→    this.save()
   837→    return true
   838→  }
   839→
   840→  count(table: TableName, filter?: Record<string, any>): number {
   841→    if (filter) return this.find(table, filter).length
   842→    return this.data[table].length
   843→  }
   844→
   845→  // Reset and reseed
   846→  reset() {
   847→    this.seed()
   848→  }
   849→}
