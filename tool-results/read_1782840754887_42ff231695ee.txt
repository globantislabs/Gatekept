    16→    16→export async function initDataService(): Promise<boolean> {
    17→    17→  if (shouldUseMockSupabase()) {

    30→    30→// Helper: try real Supabase, fall back to mock on error
    31→    31→async function tryRealOrMock<T>(realFn: () => Promise<T>, mockFn: () => T): Promise<T> {
    32→    32→  if (!_useReal) return mockFn()
    33→    33→  try {
    34→    34→    return await realFn()
    35→    35→  } catch (err) {
    36→    36→    console.warn('[DataService] Real Supabase query failed, falling back to mock:', err)
    37→    37→    return mockFn()
    38→    38→  }
    39→    39→}

   298→   298→  async getCampaignSales() {
   299→   299→    return tryRealOrMock(
   300→   300→      async () => {
   301→   301→        // For real Supabase: join qr_scans -> users -> orders
   302→   302→        const { data: scans } = await supabase.from('qr_scans').select('campaign_id, user_id')
   303→   303→        const { data: orders } = await supabase.from('orders').select('user_id, amount, status')
   304→   304→        const salesByCampaign: Record<string, { orderCount: number; revenue: number }> = {}
   305→   305→        if (scans && orders) {
   306→   306→          // Build user -> latest campaign mapping from scans
   307→   307→          const userCampaignMap: Record<string, string> = {}
   308→   308→          scans.forEach((s: any) => {
   309→   309→            if (s.user_id && s.campaign_id) userCampaignMap[s.user_id] = s.campaign_id
   310→   310→          })
   311→   311→          // Attribute orders to campaigns
   312→   312→          orders.forEach((o: any) => {
   313→   313→            const campId = userCampaignMap[o.user_id]
   314→   314→            if (campId && ['CONFIRMED', 'SHIPPED', 'DELIVERED'].includes(o.status)) {
   315→   315→              if (!salesByCampaign[campId]) salesByCampaign[campId] = { orderCount: 0, revenue: 0 }
   316→   316→              salesByCampaign[campId].orderCount++
   317→   317→              salesByCampaign[campId].revenue += o.amount || 0
   318→   318→            }
   319→   319→          })
   320→   320→        }
   321→   321→        return { data: salesByCampaign, error: null }
   322→   322→      },
   323→   323→      () => {
   324→   324→        const store = getMockStore()
   325→   325→        const scans = store.getAll<QrScan>('qr_scans')
   326→   326→        const orders = store.getAll<Order>('orders')
   327→   327→        const salesByCampaign: Record<string, { orderCount: number; revenue: number; users: string[] }> = {}
   328→   328→
   329→   329→        // Build user -> campaign mapping from scans (most recent scan wins)
   330→   330→        const userScans = scans.filter(s => s.user_id && s.campaign_id)
   331→   331→        const userCampaignMap: Record<string, string> = {}
   332→   332→        userScans.forEach(s => {
   333→   333→          userCampaignMap[s.user_id!] = s.campaign_id!
   334→   334→        })
   335→   335→
   336→   336→        // Attribute orders to campaigns
   337→   337→        orders.forEach(o => {
   338→   338→          const campId = userCampaignMap[o.user_id]
   339→   339→          if (campId && ['CONFIRMED', 'SHIPPED', 'DELIVERED'].includes(o.status)) {
   340→   340→            if (!salesByCampaign[campId]) salesByCampaign[campId] = { orderCount: 0, revenue: 0, users: [] }
   341→   341→            salesByCampaign[campId].orderCount++
   342→   342→            salesByCampaign[campId].revenue += o.amount || 0
   343→   343→            if (!salesByCampaign[campId].users.includes(o.user_id)) {
   344→   344→              salesByCampaign[campId].users.push(o.user_id)
   345→   345→            }
   346→   346→          }
   347→   347→        })
   348→   348→
   349→   349→        return { data: salesByCampaign, error: null }
   350→   350→      }
   351→   351→    )
   352→   352→  },
   353→   353→}
