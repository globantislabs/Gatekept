    37→    37→export function ProductLearningModule() {

    39→    39→
    40→    40→  const [product, setProduct] = useState<Product | null>(null)
    41→    41→  const [videos, setVideos] = useState<ProductVideo[]>([])

    48→    48→  // ─── Fetch data ────────────────────────────────────────

    53→    53→      setLoading(true)

    56→    56→          productService.getById(selectedProductId),
    57→    57→          productVideoService.getByProduct(selectedProductId),
    58→    58→          user ? productLearningService.getProgress(user.user_id, selectedProductId) : Promise.resolve({ data: null, error: null }),
    59→    59→        ])
    60→    60→        if (cancelled) return
    61→    61→        setProduct(prodRes.data || null)
    62→    62→        setVideos(vidRes.data || [])
    63→    63→        setProgress(progRes.data || null)
    64→    64→      } catch (err) {
    65→    65→        console.error('Failed to load learning data:', err)
    66→    66→      } finally {
    67→    67→        if (!cancelled) setLoading(false)
    68→    68→      }
    69→    69→    }
    70→    70→    load()
    71→    71→    return () => { cancelled = true }
    72→    72→  }, [selectedProductId, user])
    73→    73→

    78→    78→    : 0
    79→    79→

    82→    82→    setPlaying(false)
    83→    83→    setVideoProgress(currentVideoSavedProgress)

    85→    85→      clearInterval(progressIntervalRef.current)
    86→    86→      progressIntervalRef.current = null
    87→    87→    }
    88→    88→  }, [currentVideoIndex, currentVideo?.id])
    89→    89→
    90→    90→  // ─── Simulated video playback ──────────────────────────

    95→    95→          const next = Math.min(prev + 2, 100)
    96→    96→          if (next >= 100) {
    97→    97→            setPlaying(false)
    98→    98→            saveProgress(next)
    99→    99→          }
   100→   100→          return next
   101→   101→        })
   102→   102→      }, 150)
   103→   103→    }
   104→   104→    return () => {
   105→   105→      if (progressIntervalRef.current) {
   106→   106→        clearInterval(progressIntervalRef.current)
   107→   107→        progressIntervalRef.current = null
   108→   108→      }
   109→   109→    }
   110→   110→  }, [playing])
   111→   111→
   112→   112→  // ─── Save progress to backend ──────────────────────────

   114→   114→    if (!user || !selectedProductId || !currentVideo) return
   115→   115→    const updatedVP = { ...(progress?.video_progress || {}), [currentVideo.id]: pct }

   117→   117→    if (res.data) setProgress(res.data)
   118→   118→  }

   120→   120→  const handlePause = () => {
   121→   121→    setPlaying(false)
   122→   122→    saveProgress(videoProgress)
   123→   123→  }
   124→   124→

   126→   126→    const rect = e.currentTarget.getBoundingClientRect()

   128→   128→    setVideoProgress(pct)
   129→   129→    saveProgress(pct)
   130→   130→  }
   131→   131→
   132→   132→  // ─── Overall progress ──────────────────────────────────

   134→   134→    ? Math.round(videos.reduce((sum, v) => sum + (progress?.video_progress?.[v.id] ?? 0), 0) / videos.length)
   135→   135→    : 0
   136→   136→
   137→   137→  const allVideosComplete = videos.length > 0 && videos.every(v => (progress?.video_progress?.[v.id] ?? 0) >= 100)
   138→   138→
   139→   139→  // ─── Loading state ─────────────────────────────────────
   140→   140→  if (loading) {
   141→   141→    return (
   142→   142→      <div className="flex items-center justify-center min-h-[60vh]">
   143→   143→        <div className="flex flex-col items-center gap-3">
   144→   144→          <RefreshCw className="h-8 w-8 animate-spin" style={{ color: BRAND.green }} />
   145→   145→          <p className="text-sm" style={{ color: BRAND.muted }}>Loading learning content...</p>
   146→   146→        </div>
   147→   147→      </div>
   148→   148→    )
   149→   149→  }
   150→   150→
   151→   151→  if (!product || videos.length === 0) {
   152→   152→    return (
   153→   153→      <div className="flex items-center justify-center min-h-[60vh]">
   154→   154→        <Card className="max-w-md w-full" style={{ borderColor: BRAND.surface }}>
   155→   155→          <CardContent className="p-6 text-center">
   156→   156→            <AlertCircle className="h-10 w-10 mx-auto mb-3" style={{ color: BRAND.muted }} />
   157→   157→            <p className="font-heading text-lg" style={{ color: BRAND.dark }}>No learning content available</p>
   158→   158→            <p className="text-sm mt-1" style={{ color: BRAND.muted }}>This product doesn't have learning videos yet.</p>
   159→   159→            <Button
   160→   160→              className="mt-4"
   161→   161→              style={{ backgroundColor: BRAND.green }}
   162→   162→              onClick={() => navigateTo('products')}
   163→   163→            >
   164→   164→              <ArrowLeft className="h-4 w-4 mr-2" /> Back to Products
   165→   165→            </Button>
   166→   166→          </CardContent>
   167→   167→        </Card>
   168→   168→      </div>
   169→   169→    )
   170→   170→  }
   171→   171→
   172→   172→  return (
   173→   173→    <motion.div {...fadeInUp} transition={{ duration: 0.4 }} className="max-w-6xl mx-auto px-4 py-6">
   174→   174→      {/* ─── Header ────────────────────────────────────── */}
   175→   175→      <div className="flex items-center gap-3 mb-6">
   176→   176→        <Button
   177→   177→          variant="ghost"
   178→   178→          size="sm"
   179→   179→          onClick={() => navigateTo('product-detail')}
   180→   180→          className="p-2"
   181→   181→        >
   182→   182→          <ArrowLeft className="h-5 w-5" style={{ color: BRAND.dark }} />
   183→   183→        </Button>
   184→   184→        <div className="flex-1">
   185→   185→          <h1 className="font-heading text-xl sm:text-2xl font-bold" style={{ color: BRAND.dark }}>
   186→   186→            Learn: {product.name}
   187→   187→          </h1>
   188→   188→          <p className="text-sm" style={{ color: BRAND.muted }}>
   189→   189→            Complete all videos to unlock the quiz
   190→   190→          </p>
   191→   191→        </div>
   192→   192→        <Badge
   193→   193→          style={{
   194→   194→            backgroundColor: allVideosComplete ? BRAND.lime : BRAND.surface,
   195→   195→            color: allVideosComplete ? BRAND.dark : BRAND.muted,
   196→   196→          }}
   197→   197→          className="text-xs font-medium"
   198→   198→        >
   199→   199→          {overallProgress}% Complete
   200→   200→        </Badge>
   201→   201→      </div>
   202→   202→
   203→   203→      {/* ─── Overall Progress Bar ──────────────────────── */}
   204→   204→      <div className="mb-6">
   205→   205→        <Progress value={overallProgress} className="h-2" />
   206→   206→      </div>
   207→   207→
   208→   208→      {/* ─── Main Content: Video Player + Sidebar ──────── */}
   209→   209→      <div className="grid grid-cols-1 lg:grid-cols-3 gap-6">
   210→   210→        {/* ─── Video Player ────────────────────────────── */}
   211→   211→        <div className="lg:col-span-2">
   212→   212→          <motion.div {...fadeInUp} transition={{ duration: 0.3, delay: 0.1 }}>
   213→   213→            <Card className="overflow-hidden" style={{ borderColor: BRAND.surface }}>
   214→   214→              {/* Simulated video area */}
   215→   215→              <div
   216→   216→                className="relative aspect-video bg-gray-900 flex items-center justify-center cursor-pointer"
   217→   217→                onClick={() => {
   218→   218→                  if (videoProgress >= 100) return
   219→   219→                  setPlaying(!playing)
   220→   220→                }}
   221→   221→              >
   222→   222→                {/* Simulated video background */}
   223→   223→                <div className="absolute inset-0 bg-gradient-to-br from-gray-800 to-gray-900">
   224→   224→                  <div className="absolute inset-0 flex items-center justify-center opacity-10">
   225→   225→                    <div
   226→   226→                      className="text-9xl font-heading font-bold select-none"
   227→   227→                      style={{ color: BRAND.lime }}
   228→   228→                    >
   229→   229→                      {product.name.charAt(0)}
   230→   230→                    </div>
   231→   231→                  </div>
   232→   232→                </div>
   233→   233→
   234→   234→                {/* Video info overlay */}
   235→   235→                <div className="absolute top-4 left-4 z-10">
   236→   236→                  <Badge
   237→   237→                    className="text-xs"
   238→   238→                    style={{ backgroundColor: BRAND.green, color: '#fff' }}
   239→   239→                  >
   240→   240→                    Video {currentVideoIndex + 1} of {videos.length}
   241→   241→                  </Badge>
   242→   242→                </div>
   243→   243→
   244→   244→                {/* Play/Pause button */}
   245→   245→                <motion.div
   246→   246→                  whileHover={{ scale: 1.1 }}
   247→   247→                  whileTap={{ scale: 0.95 }}
   248→   248→                  className="z-10"
   249→   249→                >
   250→   250→                  {videoProgress >= 100 ? (
   251→   251→                    <div className="flex flex-col items-center gap-2">
   252→   252→                      <CheckCircle className="h-16 w-16" style={{ color: BRAND.lime }} />
   253→   253→                      <span className="text-sm font-medium" style={{ color: BRAND.lime }}>Completed</span>
   254→   254→                    </div>
   255→   255→                  ) : (
   256→   256→                    <div
   257→   257→                      className="h-16 w-16 rounded-full flex items-center justify-center backdrop-blur-sm"
   258→   258→                      style={{ backgroundColor: `${BRAND.green}CC` }}
   259→   259→                    >
   260→   260→                      {playing ? (
   261→   261→                        <div className="flex gap-1.5 items-center">
   262→   262→                          <div className="w-1.5 h-6 bg-white rounded-sm" />
   263→   263→                          <div className="w-1.5 h-6 bg-white rounded-sm" />
   264→   264→                        </div>
   265→   265→                      ) : (
   266→   266→                        <Play className="h-7 w-7 text-white ml-1" fill="white" />
   267→   267→                      )}
   268→   268→                    </div>
   269→   269→                  )}
   270→   270→                </motion.div>
   271→   271→
   272→   272→                {/* Duration badge */}
   273→   273→                <div className="absolute bottom-4 right-4 z-10">
   274→   274→                  <span className="text-xs text-white/70 bg-black/40 px-2 py-1 rounded">
   275→   275→                    {currentVideo?.duration || '0:00'}
   276→   276→                  </span>
   277→   277→                </div>
   278→   278→              </div>
   279→   279→
   280→   280→              {/* Progress bar (seek bar) */}
   281→   281→              <div
   282→   282→                className="h-1.5 cursor-pointer relative"
   283→   283→                style={{ backgroundColor: BRAND.surface }}
   284→   284→                onClick={handleSeek}
   285→   285→              >
   286→   286→                <div
   287→   287→                  className="h-full transition-all duration-300"
   288→   288→                  style={{
   289→   289→                    width: `${videoProgress}%`,
   290→   290→                    backgroundColor: videoProgress >= 100 ? BRAND.lime : BRAND.green,
   291→   291→                  }}
   292→   292→                />
   293→   293→              </div>
   294→   294→
   295→   295→              {/* Video title & controls */}
   296→   296→              <CardContent className="p-4">
   297→   297→                <h3 className="font-heading text-lg font-semibold" style={{ color: BRAND.dark }}>
   298→   298→                  {currentVideo?.title || 'Select a video'}
   299→   299→                </h3>
   300→   300→                <p className="text-sm mt-1" style={{ color: BRAND.muted }}>
   301→   301→                  {currentVideo?.description || ''}
   302→   302→                </p>
   303→   303→
   304→   304→                <div className="flex items-center justify-between mt-4">
   305→   305→                  <div className="flex items-center gap-2">
   306→   306→                    <span className="text-sm font-medium" style={{ color: BRAND.green }}>
   307→   307→                      {videoProgress}%
   308→   308→                    </span>
   309→   309→                    <span className="text-sm" style={{ color: BRAND.muted }}>
   310→   310→                      watched
   311→   311→                    </span>
   312→   312→                  </div>
   313→   313→
   314→   314→                  <div className="flex gap-2">
   315→   315→                    {currentVideoIndex > 0 && (
   316→   316→                      <Button
   317→   317→                        variant="outline"
   318→   318→                        size="sm"
   319→   319→                        onClick={() => setCurrentVideoIndex(currentVideoIndex - 1)}
   320→   320→                        style={{ borderColor: BRAND.surface, color: BRAND.dark }}
   321→   321→                      >
   322→   322→                        <ArrowLeft className="h-4 w-4 mr-1" /> Previous
   323→   323→                      </Button>
   324→   324→                    )}
   325→   325→                    {currentVideoIndex < videos.length - 1 && (
   326→   326→                      <Button
   327→   327→                        size="sm"
   328→   328→                        onClick={() => setCurrentVideoIndex(currentVideoIndex + 1)}
   329→   329→                        style={{ backgroundColor: BRAND.green, color: '#fff' }}
   330→   330→                      >
   331→   331→                        Next <ArrowRight className="h-4 w-4 ml-1" />
   332→   332→                      </Button>
   333→   333→                    )}
   334→   334→                  </div>
   335→   335→                </div>
   336→   336→              </CardContent>
   337→   337→            </Card>
   338→   338→          </motion.div>
   339→   339→
   340→   340→          {/* ─── Take Quiz CTA ─────────────────────────── */}
   341→   341→          {allVideosComplete && (
   342→   342→            <motion.div {...fadeInUp} transition={{ duration: 0.3, delay: 0.2 }} className="mt-6">
   343→   343→              <Card
   344→   344→                className="border-2"
   345→   345→                style={{ borderColor: BRAND.lime, backgroundColor: `${BRAND.lime}15` }}
   346→   346→              >
   347→   347→                <CardContent className="p-6 flex flex-col sm:flex-row items-center gap-4">
   348→   348→                  <div className="flex-1 text-center sm:text-left">
   349→   349→                    <h3 className="font-heading text-lg font-bold" style={{ color: BRAND.dark }}>
   350→   350→                      All Videos Complete! 🎉
   351→   351→                    </h3>
   352→   352→                    <p className="text-sm" style={{ color: BRAND.muted }}>
   353→   353→                      You're ready to take the quiz for {product.name}
   354→   354→                    </p>
   355→   355→                  </div>
   356→   356→                  <Button
   357→   357→                    size="lg"
   358→   358→                    className="font-heading font-semibold"
   359→   359→                    style={{ backgroundColor: BRAND.green, color: '#fff' }}
   360→   360→                    onClick={() => {
   361→   361→                      setCurrentQuizIndex(0)
   362→   362→                      navigateTo('product-quiz')
   363→   363→                    }}
   364→   364→                  >
   365→   365→                    Take Quiz <ChevronRight className="h-5 w-5 ml-1" />
   366→   366→                  </Button>
   367→   367→                </CardContent>
   368→   368→              </Card>
   369→   369→            </motion.div>
   370→   370→          )}

   372→   372→
   373→   373→        {/* ─── Video List Sidebar ──────────────────────── */}
   374→   374→        <div className="lg:col-span-1">
   375→   375→          <Card style={{ borderColor: BRAND.surface }}>

   378→   378→                Video Lessons
   379→   379→              </CardTitle>
   380→   380→              <CardDescription className="text-xs" style={{ color: BRAND.muted }}>

   384→   384→            <CardContent className="p-0">

   435→   435→
