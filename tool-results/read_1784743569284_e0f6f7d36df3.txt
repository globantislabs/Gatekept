     1→'use client'
     2→import React, { useState, useEffect, useRef, useCallback } from 'react'
     3→import { motion, AnimatePresence } from 'framer-motion'
     4→import {
     5→  Play, CheckCircle, Lock, ArrowRight, ArrowLeft, RefreshCw,
     6→  AlertCircle, ChevronRight, Trophy, Video, FileQuestion, RotateCcw,
     7→} from 'lucide-react'
     8→import { useAppStore } from '@/store/app-store'
     9→import { productVideoService, productQuizService, productLearningService, productService } from '@/lib/data-service'
    10→import type { ProductVideo, ProductQuiz, ProductLearningProgress, Product } from '@/lib/data-service'
    11→import { Button } from '@/components/ui/button'
    12→import { Card, CardContent, CardHeader, CardTitle, CardDescription, CardFooter } from '@/components/ui/card'
    13→import { Badge } from '@/components/ui/badge'
    14→import { Progress } from '@/components/ui/progress'
    15→import { RadioGroup, RadioGroupItem } from '@/components/ui/radio-group'
    16→import { Label } from '@/components/ui/label'
    17→import { toast } from 'sonner'
    18→
    19→// ─── Brand Constants ───────────────────────────────────────
    20→const BRAND = {
    21→  green: '#48805b',
    22→  lime: '#afb75d',
    23→  dark: '#1f1e1c',
    24→  muted: '#88837b',
    25→  surface: '#e3dfd8',
    26→  bg: '#f4f3f0',
    27→  blue: '#2e91b2',
    28→} as const
    29→
    30→const PASS_THRESHOLD = 4 // need 4 out of 5 correct (80%)
    31→
    32→// ─── Step Types ────────────────────────────────────────────
    33→type LearningStep =
    34→  | { type: 'video'; videoIndex: number }
    35→  | { type: 'quiz'; videoIndex: number }
    36→  | { type: 'completed' }
    37→
    38→// ─── Animation Helpers ─────────────────────────────────────
    39→const fadeInUp = {
    40→  initial: { opacity: 0, y: 20 } as const,
    41→  animate: { opacity: 1, y: 0 } as const,
    42→}
    43→
    44→// ─── Quiz result for a specific video ──────────────────────
    45→interface QuizResult {
    46→  correct: number
    47→  total: number
    48→  passed: boolean
    49→  answers: Record<number, number>
    50→}
    51→
    52→// ═══════════════════════════════════════════════════════════
    53→// ProductLearningModule — Unified sequential learning flow
    54→// Video 1 → Quiz 1 → Video 2 → Quiz 2 → Video 3 → Quiz 3 → Unlocked
    55→// ═══════════════════════════════════════════════════════════
    56→export function ProductLearningModule() {
    57→  const { selectedProductId, user, navigateTo } = useAppStore()
    58→
    59→  // ─── Data state ───────────────────────────────────────
    60→  const [product, setProduct] = useState<Product | null>(null)
    61→  const [videos, setVideos] = useState<ProductVideo[]>([])
    62→  const [progress, setProgress] = useState<ProductLearningProgress | null>(null)
    63→  const [loading, setLoading] = useState(true)
    64→
    65→  // ─── Learning step state ──────────────────────────────
    66→  const [currentStep, setCurrentStep] = useState<LearningStep>({ type: 'video', videoIndex: 0 })
    67→  const [passedQuizzes, setPassedQuizzes] = useState<Record<number, boolean>>({})
    68→
    69→  // ─── Video player state ───────────────────────────────
    70→  const [playing, setPlaying] = useState(false)
    71→  const [videoProgress, setVideoProgress] = useState(0) // 0-100 for current video
    72→  const progressIntervalRef = useRef<ReturnType<typeof setInterval> | null>(null)
    73→
    74→  // ─── Quiz state ───────────────────────────────────────
    75→  const [quizQuestions, setQuizQuestions] = useState<ProductQuiz[]>([])
    76→  const [quizLoading, setQuizLoading] = useState(false)
    77→  const [currentQuestionIdx, setCurrentQuestionIdx] = useState(0)
    78→  const [answers, setAnswers] = useState<Record<number, number>>({})
    79→  const [quizResult, setQuizResult] = useState<QuizResult | null>(null)
    80→  const [submitting, setSubmitting] = useState(false)
    81→
    82→  // ─── Derived ──────────────────────────────────────────
    83→  const currentVideo = currentStep.type === 'video' || currentStep.type === 'quiz'
    84→    ? videos[currentStep.videoIndex] || null
    85→    : null
    86→
    87→  // ─── Build the step list for the progress indicator ───
    88→  const stepList: { type: 'video' | 'quiz'; videoIndex: number; label: string }[] = []
    89→  for (let i = 0; i < videos.length; i++) {
    90→    stepList.push({ type: 'video', videoIndex: i, label: `Video ${i + 1}` })
    91→    stepList.push({ type: 'quiz', videoIndex: i, label: `Quiz ${i + 1}` })
    92→  }
    93→  const currentStepIndex = stepList.findIndex((s) => {
    94→    if (currentStep.type === 'completed') return false
    95→    return s.type === currentStep.type && s.videoIndex === currentStep.videoIndex
    96→  })
    97→
    98→  // ─── Fetch data ───────────────────────────────────────
    99→  useEffect(() => {
   100→    if (!selectedProductId) return
   101→    let cancelled = false
   102→    async function load() {
   103→      setLoading(true)
   104→      try {
   105→        const [prodRes, vidRes, progRes] = await Promise.all([
   106→          productService.getById(selectedProductId),
   107→          productVideoService.getByProduct(selectedProductId),
   108→          user
   109→            ? productLearningService.getProgress(user.user_id, selectedProductId)
   110→            : Promise.resolve({ data: null, error: null }),
   111→        ])
   112→        if (cancelled) return
   113→        setProduct(prodRes.data || null)
   114→        const loadedVideos = vidRes.data || []
   115→        setVideos(loadedVideos)
   116→        setProgress(progRes.data || null)
   117→
   118→        // Determine which step to start on based on saved progress
   119→        if (progRes.data && loadedVideos.length > 0) {
   120→          const savedProgress = progRes.data as ProductLearningProgress
   121→          // Check quiz_answers to see which quizzes were already passed
   122→          // We'll reconstruct the step from the progress
   123→          const vp = savedProgress.video_progress || {}
   124→          const qa = savedProgress.quiz_answers || {}
   125→
   126→          // Find the first video that is not 100% complete
   127→          let startVideoIdx = 0
   128→          for (let i = 0; i < loadedVideos.length; i++) {
   129→            const vidProg = vp[loadedVideos[i].id] ?? 0
   130→            if (vidProg < 100) {
   131→              startVideoIdx = i
   132→              break
   133→            }
   134→            if (i === loadedVideos.length - 1) {
   135→              startVideoIdx = loadedVideos.length - 1
   136→            }
   137→          }
   138→
   139→          // If the status is already UNLOCKED or COMPLETED, go to completed
   140→          if (savedProgress.status === 'UNLOCKED' || savedProgress.status === 'COMPLETED') {
   141→            const allPassed: Record<number, boolean> = {}
   142→            for (let i = 0; i < loadedVideos.length; i++) {
   143→              allPassed[i] = true
   144→            }
   145→            setPassedQuizzes(allPassed)
   146→            setCurrentStep({ type: 'completed' })
   147→          } else {
   148→            // Start on the appropriate video
   149→            setCurrentStep({ type: 'video', videoIndex: startVideoIdx })
   150→          }
   151→        }
   152→      } catch (err) {
   153→        console.error('Failed to load learning data:', err)
   154→      } finally {
   155→        if (!cancelled) setLoading(false)
   156→      }
   157→    }
   158→    load()
   159→    return () => { cancelled = true }
   160→  }, [selectedProductId, user])
   161→
   162→  // ─── Reset video player when step changes ─────────────
   163→  useEffect(() => {
   164→    setPlaying(false)
   165→    setQuizResult(null)
   166→    setAnswers({})
   167→    setCurrentQuestionIdx(0)
   168→    if (progressIntervalRef.current) {
   169→      clearInterval(progressIntervalRef.current)
   170→      progressIntervalRef.current = null
   171→    }
   172→
   173→    // Set video progress from saved data
   174→    if (currentStep.type === 'video' && currentVideo) {
   175→      const savedProg = progress?.video_progress?.[currentVideo.id] ?? 0
   176→      setVideoProgress(savedProg)
   177→    } else {
   178→      setVideoProgress(0)
   179→    }
   180→  }, [currentStep.type, currentStep.type === 'video' || currentStep.type === 'quiz' ? currentStep.videoIndex : null, currentVideo?.id])
   181→
   182→  // ─── Fetch quiz questions when entering a quiz step ───
   183→  useEffect(() => {
   184→    if (currentStep.type !== 'quiz') return
   185→    const video = videos[currentStep.videoIndex]
   186→    if (!video) return
   187→
   188→    let cancelled = false
   189→    async function loadQuiz() {
   190→      setQuizLoading(true)
   191→      try {
   192→        const res = await productQuizService.getByVideo(video.id)
   193→        if (!cancelled) {
   194→          setQuizQuestions(res.data || [])
   195→        }
   196→      } catch (err) {
   197→        console.error('Failed to load quiz questions:', err)
   198→      } finally {
   199→        if (!cancelled) setQuizLoading(false)
   200→      }
   201→    }
   202→    loadQuiz()
   203→    return () => { cancelled = true }
   204→  }, [currentStep.type === 'quiz' ? currentStep.videoIndex : -1, videos])
   205→
   206→  // ─── Simulated video playback ─────────────────────────
   207→  useEffect(() => {
   208→    if (playing && videoProgress < 100) {
   209→      progressIntervalRef.current = setInterval(() => {
   210→        setVideoProgress((prev) => {
   211→          const next = Math.min(prev + 2, 100)
   212→          if (next >= 100) {
   213→            setPlaying(false)
   214→          }
   215→          return next
   216→        })
   217→      }, 150)
   218→    }
   219→    return () => {
   220→      if (progressIntervalRef.current) {
   221→        clearInterval(progressIntervalRef.current)
   222→        progressIntervalRef.current = null
   223→      }
   224→    }
   225→  }, [playing])
   226→
   227→  // ─── Save video progress to backend ───────────────────
   228→  const saveVideoProgress = useCallback(async (pct: number) => {
   229→    if (!selectedProductId || !currentVideo) return
   230→    if (!user) return // Skip backend save if not logged in
   231→    const updatedVP = { ...(progress?.video_progress || {}), [currentVideo.id]: pct }
   232→    const res = await productLearningService.updateVideoProgress(user.user_id, selectedProductId, updatedVP)
   233→    if (res.data) setProgress(res.data)
   234→  }, [user, selectedProductId, currentVideo, progress?.video_progress])
   235→
   236→  // ─── Auto-save when video completes ───────────────────
   237→  useEffect(() => {
   238→    if (videoProgress >= 100 && currentStep.type === 'video' && currentVideo) {
   239→      saveVideoProgress(100)
   240→    }
   241→  }, [videoProgress, currentStep.type, currentVideo?.id, saveVideoProgress])
   242→
   243→  // ─── Video player handlers ────────────────────────────
   244→  const handlePlayPause = () => {
   245→    if (videoProgress >= 100) return
   246→    setPlaying(!playing)
   247→    if (playing) {
   248→      saveVideoProgress(videoProgress)
   249→    }
   250→  }
   251→
   252→  const handleSeek = (e: React.MouseEvent<HTMLDivElement>) => {
   253→    const rect = e.currentTarget.getBoundingClientRect()
   254→    const pct = Math.round(((e.clientX - rect.left) / rect.width) * 100)
   255→    setVideoProgress(pct)
   256→    saveVideoProgress(pct)
   257→  }
   258→
   259→  // ─── Advance to next step after video completes ───────
   260→  const handleVideoComplete = () => {
   261→    if (currentStep.type !== 'video') return
   262→    // Move to quiz for this video
   263→    setCurrentStep({ type: 'quiz', videoIndex: currentStep.videoIndex })
   264→  }
   265→
   266→  // ─── Handle answer selection ──────────────────────────
   267→  const handleAnswer = (questionIndex: number, optionIndex: number) => {
   268→    setAnswers((prev) => ({ ...prev, [questionIndex]: optionIndex }))
   269→  }
   270→
   271→  // ─── Submit quiz ──────────────────────────────────────
   272→  const handleSubmitQuiz = async () => {
   273→    if (!selectedProductId) return
   274→    setSubmitting(true)
   275→
   276→    try {
   277→      let correctCount = 0
   278→      const quizAnswersMap: Record<string, number> = {}
   279→
   280→      quizQuestions.forEach((quiz, idx) => {
   281→        const userAnswer = answers[idx]
   282→        quizAnswersMap[quiz.id] = userAnswer ?? -1
   283→        if (userAnswer === quiz.answer) {
   284→          correctCount++
   285→        }
   286→      })
   287→
   288→      const totalQuestions = quizQuestions.length
   289→      const passed = correctCount >= PASS_THRESHOLD
   290→
   291→      const result: QuizResult = {
   292→        correct: correctCount,
   293→        total: totalQuestions,
   294→        passed,
   295→        answers: { ...answers },
   296→      }
   297→      setQuizResult(result)
   298→
   299→      // Submit via service (only if logged in — otherwise just check locally)
   300→      if (user) {
   301→        const overallScore = totalQuestions > 0 ? Math.round((correctCount / totalQuestions) * 100) : 0
   302→        const res = await productLearningService.submitQuiz(
   303→          user.user_id,
   304→          selectedProductId,
   305→          quizAnswersMap,
   306→          overallScore,
   307→          passed
   308→        )
   309→
   310→        if (res.data) {
   311→          setProgress(res.data)
   312→        }
   313→      }
   314→
   315→      if (passed) {
   316→        // Mark this quiz as passed
   317→        setPassedQuizzes((prev) => ({
   318→          ...prev,
   319→          [currentStep.type === 'quiz' ? currentStep.videoIndex : -1]: true,
   320→        }))
   321→        toast.success('Quiz passed! Moving to the next lesson.')
   322→      } else {
   323→        toast.error(`You got ${correctCount}/${totalQuestions} correct. You need ${PASS_THRESHOLD} to pass. Please re-watch the video and try again.`)
   324→      }
   325→    } catch (err) {
   326→      console.error('Quiz submit failed:', err)
   327→      toast.error('Failed to submit quiz. Please try again.')
   328→    } finally {
   329→      setSubmitting(false)
   330→    }
   331→  }
   332→
   333→  // ─── Advance after passing quiz ───────────────────────
   334→  const handleQuizPassContinue = () => {
   335→    if (currentStep.type !== 'quiz') return
   336→    const nextVideoIndex = currentStep.videoIndex + 1
   337→    if (nextVideoIndex < videos.length) {
   338→      setCurrentStep({ type: 'video', videoIndex: nextVideoIndex })
   339→    } else {
   340→      // All quizzes passed — unlock the product
   341→      setCurrentStep({ type: 'completed' })
   342→    }
   343→  }
   344→
   345→  // ─── Retry after failing quiz ─────────────────────────
   346→  const handleRetryQuiz = () => {
   347→    if (currentStep.type !== 'quiz') return
   348→    // Go back to the video for this quiz
   349→    setCurrentStep({ type: 'video', videoIndex: currentStep.videoIndex })
   350→  }
   351→
   352→  // ─── Check if a step is unlocked ──────────────────────
   353→  const isStepUnlocked = (step: { type: 'video' | 'quiz'; videoIndex: number }): boolean => {
   354→    // Step 0 (Video 1) is always unlocked
   355→    if (step.videoIndex === 0 && step.type === 'video') return true
   356→    // Quiz for video 0 is unlocked if video 0 is 100% complete
   357→    if (step.videoIndex === 0 && step.type === 'quiz') {
   358→      return (progress?.video_progress?.[videos[0]?.id] ?? 0) >= 100
   359→    }
   360→    // Video N is unlocked if quiz for video N-1 is passed
   361→    if (step.type === 'video' && step.videoIndex > 0) {
   362→      return !!passedQuizzes[step.videoIndex - 1]
   363→    }
   364→    // Quiz N is unlocked if video N is 100% complete
   365→    if (step.type === 'quiz' && step.videoIndex > 0) {
   366→      const videoComplete = (progress?.video_progress?.[videos[step.videoIndex]?.id] ?? 0) >= 100
   367→      const prevQuizPassed = !!passedQuizzes[step.videoIndex - 1]
   368→      return videoComplete && prevQuizPassed
   369→    }
   370→    return false
   371→  }
   372→
   373→  // ─── Overall progress percentage ──────────────────────
   374→  const totalSteps = videos.length * 2 // each video + quiz
   375→  const completedSteps = Object.keys(passedQuizzes).filter((k) => passedQuizzes[Number(k)]).length * 2
   376→    + videos.filter((v) => (progress?.video_progress?.[v.id] ?? 0) >= 100 && !passedQuizzes[videos.indexOf(v)]).length
   377→  const overallPct = totalSteps > 0 ? Math.round((completedSteps / totalSteps) * 100) : 0
   378→
   379→  // ─── Loading state ─────────────────────────────────────
   380→  if (loading) {
   381→    return (
   382→      <div className="flex items-center justify-center min-h-[60vh]">
   383→        <div className="flex flex-col items-center gap-3">
   384→          <RefreshCw className="h-8 w-8 animate-spin" style={{ color: BRAND.green }} />
   385→          <p className="text-sm" style={{ color: BRAND.muted }}>Loading learning content...</p>
   386→        </div>
   387→      </div>
   388→    )
   389→  }
   390→
   391→  if (!product || videos.length === 0) {
   392→    return (
   393→      <div className="flex items-center justify-center min-h-[60vh]">
   394→        <Card className="max-w-md w-full" style={{ borderColor: BRAND.surface }}>
   395→          <CardContent className="p-6 text-center">
   396→            <AlertCircle className="h-10 w-10 mx-auto mb-3" style={{ color: BRAND.muted }} />
   397→            <p className="font-heading text-lg" style={{ color: BRAND.dark }}>No learning content available</p>
   398→            <p className="text-sm mt-1" style={{ color: BRAND.muted }}>This product doesn't have learning videos yet.</p>
   399→            <Button
   400→              className="mt-4"
   401→              style={{ backgroundColor: BRAND.green, color: '#fff' }}
   402→              onClick={() => navigateTo('products')}
   403→            >
   404→              <ArrowLeft className="h-4 w-4 mr-2" /> Back to Products
   405→            </Button>
   406→          </CardContent>
   407→        </Card>
   408→      </div>
   409→    )
   410→  }
   411→
   412→  // ═══════════════════════════════════════════════════════
   413→  // RENDER
   414→  // ═══════════════════════════════════════════════════════
   415→  return (
   416→    <motion.div {...fadeInUp} transition={{ duration: 0.4 }} className="max-w-4xl mx-auto px-4 py-6">
   417→      {/* ─── Header ────────────────────────────────────── */}
   418→      <div className="flex items-center gap-3 mb-4">
   419→        <Button
   420→          variant="ghost"
   421→          size="sm"
   422→          onClick={() => navigateTo('products')}
   423→          className="p-2"
   424→        >
   425→          <ArrowLeft className="h-5 w-5" style={{ color: BRAND.dark }} />
   426→        </Button>
   427→        <div className="flex-1">
   428→          <h1 className="font-heading text-xl sm:text-2xl font-bold" style={{ color: BRAND.dark }}>
   429→            Learn: {product.name}
   430→          </h1>
   431→          <p className="text-sm" style={{ color: BRAND.muted }}>
   432→            Complete each video and quiz to unlock this product
   433→          </p>
   434→        </div>
   435→        <Badge
   436→          style={{
   437→            backgroundColor: currentStep.type === 'completed' ? BRAND.lime : BRAND.surface,
   438→            color: currentStep.type === 'completed' ? BRAND.dark : BRAND.muted,
   439→          }}
   440→          className="text-xs font-medium"
   441→        >
   442→          {currentStep.type === 'completed' ? 'Unlocked' : `${overallPct}% Progress`}
   443→        </Badge>
   444→      </div>
   445→
   446→      {/* ─── Step Progress Indicator ───────────────────── */}
   447→      <Card className="mb-6" style={{ borderColor: BRAND.surface }}>
   448→        <CardContent className="p-4">
   449→          <div className="flex items-center justify-between gap-1 overflow-x-auto pb-1">
   450→            {stepList.map((step, idx) => {
   451→              const isCurrent =
   452→                currentStep.type !== 'completed' &&
   453→                step.type === currentStep.type &&
   454→                step.videoIndex === currentStep.videoIndex
   455→              const isPast = idx < currentStepIndex || currentStep.type === 'completed'
   456→              const isUnlocked = isStepUnlocked(step)
   457→              const isQuizPassed = step.type === 'quiz' && !!passedQuizzes[step.videoIndex]
   458→              const isVideoComplete =
   459→                step.type === 'video' &&
   460→                (progress?.video_progress?.[videos[step.videoIndex]?.id] ?? 0) >= 100
   461→
   462→              // Determine icon and color
   463→              let stepIcon: React.ReactNode
   464→              let bgColor: string
   465→              let iconColor: string
   466→              let textColor: string
   467→
   468→              if (isQuizPassed || (isPast && step.type === 'quiz')) {
   469→                stepIcon = <CheckCircle className="h-4 w-4" />
   470→                bgColor = BRAND.lime
   471→                iconColor = BRAND.dark
   472→                textColor = BRAND.dark
   473→              } else if (isPast && step.type === 'video') {
   474→                stepIcon = <CheckCircle className="h-4 w-4" />
   475→                bgColor = BRAND.lime
   476→                iconColor = BRAND.dark
   477→                textColor = BRAND.dark
   478→              } else if (isVideoComplete && step.type === 'video') {
   479→                stepIcon = <CheckCircle className="h-4 w-4" />
   480→                bgColor = BRAND.lime
   481→                iconColor = BRAND.dark
   482→                textColor = BRAND.dark
   483→              } else if (isCurrent) {
   484→                stepIcon =
   485→                  step.type === 'video' ? (
   486→                    <Play className="h-4 w-4" fill="white" />
   487→                  ) : (
   488→                    <FileQuestion className="h-4 w-4" />
   489→                  )
   490→                bgColor = BRAND.green
   491→                iconColor = '#fff'
   492→                textColor = BRAND.green
   493→              } else if (isUnlocked) {
   494→                stepIcon =
   495→                  step.type === 'video' ? (
   496→                    <Video className="h-4 w-4" />
   497→                  ) : (
   498→                    <FileQuestion className="h-4 w-4" />
   499→                  )
   500→                bgColor = `${BRAND.green}20`
   501→                iconColor = BRAND.green
   502→                textColor = BRAND.green
   503→              } else {
   504→                stepIcon = <Lock className="h-3.5 w-3.5" />
   505→                bgColor = BRAND.surface
   506→                iconColor = BRAND.muted
   507→                textColor = BRAND.muted
   508→              }
   509→
   510→              return (
   511→                <React.Fragment key={`${step.type}-${step.videoIndex}`}>
   512→                  {idx > 0 && (
   513→                    <div
   514→                      className="h-0.5 flex-1 min-w-[8px] max-w-[32px] rounded-full"
   515→                      style={{
   516→                        backgroundColor:
   517→                          isPast || isCurrent ? BRAND.lime : BRAND.surface,
   518→                      }}
   519→                    />
   520→                  )}
   521→                  <div className="flex flex-col items-center gap-1 min-w-[48px]">
   522→                    <div
   523→                      className="h-8 w-8 rounded-full flex items-center justify-center transition-all"
   524→                      style={{ backgroundColor: bgColor, color: iconColor }}
   525→                    >
   526→                      {stepIcon}
   527→                    </div>
   528→                    <span
   529→                      className="text-[10px] font-medium whitespace-nowrap"
   530→                      style={{ color: textColor }}
   531→                    >
