     1→'use client'
     2→import React, { useState, useEffect, useRef, useCallback } from 'react'
     3→import { motion, AnimatePresence } from 'framer-motion'
     4→import {
     5→  Play, CheckCircle, Lock, ArrowRight, ArrowLeft, RefreshCw,
     6→  AlertCircle, ChevronRight, Trophy, Video, FileQuestion, RotateCcw,
     7→} from 'lucide-react'
     8→import { useAppStore } from '@/store/app-store'
     9→import { productVideoService, productQuizService, productLearningService, productService, productServiceCompat, productVideoServiceCompat, productLearningServiceCompat } from '@/lib/data-service'
    10→import type { ProductVideo, ProductQuiz, ProductLearningProgress, Product } from '@/lib/data-service'
    11→import { Button } from '@/components/ui/button'
    12→import { Card, CardContent, CardHeader, CardTitle, CardDescription, CardFooter } from '@/components/ui/card'
    13→import { Badge } from '@/components/ui/badge'
    14→import { Progress } from '@/components/ui/progress'
    15→import { RadioGroup, RadioGroupItem } from '@/components/ui/radio-group'
    16→import { Label } from '@/components/ui/label'
    17→import { toast } from 'sonner'
    18→
    19→// ─── Brand Constants ───────────────────────────────────────
    20→const BRAND = {
    21→  green: '#48805b',
    22→  lime: '#afb75d',
    23→  dark: '#1f1e1c',
    24→  muted: '#88837b',
    25→  surface: '#e3dfd8',
    26→  bg: '#f4f3f0',
    27→  blue: '#2e91b2',
    28→} as const
    29→
    30→const PASS_THRESHOLD = 4 // need 4 out of 5 correct (80%)
    31→
    32→// ─── Step Types ────────────────────────────────────────────
    33→type LearningStep =
    34→  | { type: 'video'; videoIndex: number }
    35→  | { type: 'quiz'; videoIndex: number }
    36→  | { type: 'completed' }
    37→
    38→// ─── Animation Helpers ─────────────────────────────────────
    39→const fadeInUp = {
    40→  initial: { opacity: 0, y: 20 } as const,
    41→  animate: { opacity: 1, y: 0 } as const,
    42→}
    43→
    44→// ─── Quiz result for a specific video ──────────────────────
    45→interface QuizResult {
    46→  correct: number
    47→  total: number
    48→  passed: boolean
    49→  answers: Record<number, number>
    50→}
    51→
    52→// ═══════════════════════════════════════════════════════════
    53→// ProductLearningModule — Unified sequential learning flow
    54→// Video 1 → Quiz 1 → Video 2 → Quiz 2 → Video 3 → Quiz 3 → Unlocked
    55→// ═══════════════════════════════════════════════════════════
    56→export function ProductLearningModule() {
    57→  const { selectedProductId, user, navigateTo } = useAppStore()
    58→
    59→  // ─── Data state ───────────────────────────────────────
    60→  const [product, setProduct] = useState<Product | null>(null)
    61→  const [videos, setVideos] = useState<ProductVideo[]>([])
    62→  const [progress, setProgress] = useState<ProductLearningProgress | null>(null)
    63→  const [loading, setLoading] = useState(true)
    64→
    65→  // ─── Learning step state ──────────────────────────────
    66→  const [currentStep, setCurrentStep] = useState<LearningStep>({ type: 'video', videoIndex: 0 })
    67→  const [passedQuizzes, setPassedQuizzes] = useState<Record<number, boolean>>({})
    68→
    69→  // ─── Video player state ───────────────────────────────
    70→  const [playing, setPlaying] = useState(false)
    71→  const [videoProgress, setVideoProgress] = useState(0) // 0-100 for current video
    72→  const progressIntervalRef = useRef<ReturnType<typeof setInterval> | null>(null)
    73→
    74→  // ─── Quiz state ───────────────────────────────────────
    75→  const [quizQuestions, setQuizQuestions] = useState<ProductQuiz[]>([])
    76→  const [quizLoading, setQuizLoading] = useState(false)
    77→  const [currentQuestionIdx, setCurrentQuestionIdx] = useState(0)
    78→  const [answers, setAnswers] = useState<Record<number, number>>({})
    79→  const [quizResult, setQuizResult] = useState<QuizResult | null>(null)
    80→  const [submitting, setSubmitting] = useState(false)
    81→
    82→  // ─── Derived ──────────────────────────────────────────
    83→  const currentVideo = currentStep.type === 'video' || currentStep.type === 'quiz'
    84→    ? videos[currentStep.videoIndex] || null
    85→    : null
    86→
    87→  // ─── Build the step list for the progress indicator ───
    88→  const stepList: { type: 'video' | 'quiz'; videoIndex: number; label: string }[] = []
    89→  for (let i = 0; i < videos.length; i++) {
    90→    stepList.push({ type: 'video', videoIndex: i, label: `Video ${i + 1}` })
    91→    stepList.push({ type: 'quiz', videoIndex: i, label: `Quiz ${i + 1}` })
    92→  }
    93→  const currentStepIndex = stepList.findIndex((s) => {
    94→    if (currentStep.type === 'completed') return false
    95→    return s.type === currentStep.type && s.videoIndex === currentStep.videoIndex
    96→  })
    97→
    98→  // ─── Fetch data ───────────────────────────────────────
    99→  useEffect(() => {
   100→    if (!selectedProductId) return
   101→    let cancelled = false
   102→    async function load() {
   103→      setLoading(true)
   104→      try {
   105→        const [prodRes, vidRes, progRes] = await Promise.all([
   106→          productServiceCompat.getById(selectedProductId),
   107→          productVideoServiceCompat.getByProduct(selectedProductId),
   108→          user
   109→            ? productLearningServiceCompat.getProgress(user.id, selectedProductId)
   110→            : Promise.resolve({ data: null, error: null }),
   111→        ])
   112→        if (cancelled) return
   113→        setProduct(prodRes.data || null)
   114→        const loadedVideos = vidRes.data || []
   115→        setVideos(loadedVideos)
   116→        setProgress(progRes.data || null)
   117→
   118→        // Determine which step to start on based on saved progress
   119→        if (progRes.data && loadedVideos.length > 0) {
   120→          const savedProgress = progRes.data as ProductLearningProgress
   121→          // Check quiz_answers to see which quizzes were already passed
   122→          // We'll reconstruct the step from the progress
   123→          const vp = savedProgress.video_progress || {}
   124→          const qa = savedProgress.quiz_answers || {}
   125→
   126→          // Find the first video that is not 100% complete
   127→          let startVideoIdx = 0
   128→          for (let i = 0; i < loadedVideos.length; i++) {
   129→            const vidProg = vp[loadedVideos[i].id] ?? 0
   130→            if (vidProg < 100) {
   131→              startVideoIdx = i
   132→              break
   133→            }
   134→            if (i === loadedVideos.length - 1) {
   135→              startVideoIdx = loadedVideos.length - 1
   136→            }
   137→          }
   138→
   139→          // If the status is already UNLOCKED or COMPLETED, go to completed
   140→          if (savedProgress.status === 'UNLOCKED' || savedProgress.status === 'COMPLETED') {
   141→            const allPassed: Record<number, boolean> = {}
   142→            for (let i = 0; i < loadedVideos.length; i++) {
   143→              allPassed[i] = true
   144→            }
   145→            setPassedQuizzes(allPassed)
   146→            setCurrentStep({ type: 'completed' })
   147→          } else {
   148→            // Start on the appropriate video
   149→            setCurrentStep({ type: 'video', videoIndex: startVideoIdx })
   150→          }
   151→        }
   152→      } catch (err) {
   153→        console.error('Failed to load learning data:', err)
   154→      } finally {
   155→        if (!cancelled) setLoading(false)
   156→      }
   157→    }
   158→    load()
   159→    return () => { cancelled = true }
   160→  }, [selectedProductId, user])
   161→
   162→  // ─── Reset video player when step changes ─────────────
   163→  useEffect(() => {
   164→    setPlaying(false)
   165→    setQuizResult(null)
   166→    setAnswers({})
   167→    setCurrentQuestionIdx(0)
   168→    if (progressIntervalRef.current) {
   169→      clearInterval(progressIntervalRef.current)
   170→      progressIntervalRef.current = null
   171→    }
   172→
   173→    // Set video progress from saved data
   174→    if (currentStep.type === 'video' && currentVideo) {
   175→      const savedProg = progress?.video_progress?.[currentVideo.id] ?? 0
   176→      setVideoProgress(savedProg)
   177→    } else {
   178→      setVideoProgress(0)
   179→    }
   180→  }, [currentStep.type, currentStep.type === 'video' || currentStep.type === 'quiz' ? currentStep.videoIndex : null, currentVideo?.id])
   181→
   182→  // ─── Fetch quiz questions when entering a quiz step ───
   183→  useEffect(() => {
   184→    if (currentStep.type !== 'quiz') return
   185→    const video = videos[currentStep.videoIndex]
   186→    if (!video) return
   187→
   188→    let cancelled = false
   189→    async function loadQuiz() {
   190→      setQuizLoading(true)
   191→      try {
   192→        const res = await productQuizService.list(selectedProductId)
   193→        if (!cancelled) {
   194→          // Filter quizzes for this specific video
   195→          const filtered = res.filter(q => q.video_id === video.id)
   196→          setQuizQuestions(filtered)
   197→        }
   198→      } catch (err) {
   199→        console.error('Failed to load quiz questions:', err)
   200→      } finally {
   201→        if (!cancelled) setQuizLoading(false)
   202→      }
   203→    }
   204→    loadQuiz()
   205→    return () => { cancelled = true }
   206→  }, [currentStep.type === 'quiz' ? currentStep.videoIndex : -1, videos])
   207→
   208→  // ─── Simulated video playback ─────────────────────────
   209→  useEffect(() => {
   210→    if (playing && videoProgress < 100) {
   211→      progressIntervalRef.current = setInterval(() => {
   212→        setVideoProgress((prev) => {
   213→          const next = Math.min(prev + 2, 100)
   214→          if (next >= 100) {
   215→            setPlaying(false)
   216→          }
   217→          return next
   218→        })
   219→      }, 150)
   220→    }
   221→    return () => {
   222→      if (progressIntervalRef.current) {
   223→        clearInterval(progressIntervalRef.current)
   224→        progressIntervalRef.current = null
   225→      }
   226→    }
   227→  }, [playing])
   228→
   229→  // ─── Save video progress to backend ───────────────────
   230→  const saveVideoProgress = useCallback(async (pct: number) => {
   231→    if (!selectedProductId || !currentVideo) return
   232→    if (!user) return // Skip backend save if not logged in
   233→    const updatedVP = { ...(progress?.video_progress || {}), [currentVideo.id]: pct }
   234→    const res = await productLearningService.updateVideoProgress(user.user_id, selectedProductId, updatedVP)
   235→    if (res.data) setProgress(res.data)
   236→  }, [user, selectedProductId, currentVideo, progress?.video_progress])
   237→
   238→  // ─── Auto-save when video completes ───────────────────
   239→  useEffect(() => {
   240→    if (videoProgress >= 100 && currentStep.type === 'video' && currentVideo) {
   241→      saveVideoProgress(100)
   242→    }
   243→  }, [videoProgress, currentStep.type, currentVideo?.id, saveVideoProgress])
   244→
   245→  // ─── Video player handlers ────────────────────────────
   246→  const handlePlayPause = () => {
   247→    if (videoProgress >= 100) return
   248→    setPlaying(!playing)
   249→    if (playing) {
   250→      saveVideoProgress(videoProgress)
   251→    }
   252→  }
   253→
   254→  const handleSeek = (e: React.MouseEvent<HTMLDivElement>) => {
   255→    const rect = e.currentTarget.getBoundingClientRect()
   256→    const pct = Math.round(((e.clientX - rect.left) / rect.width) * 100)
   257→    setVideoProgress(pct)
   258→    saveVideoProgress(pct)
   259→  }
   260→
   261→  // ─── Advance to next step after video completes ───────
   262→  const handleVideoComplete = () => {
   263→    if (currentStep.type !== 'video') return
   264→    // Move to quiz for this video
   265→    setCurrentStep({ type: 'quiz', videoIndex: currentStep.videoIndex })
   266→  }
   267→
   268→  // ─── Handle answer selection ──────────────────────────
   269→  const handleAnswer = (questionIndex: number, optionIndex: number) => {
   270→    setAnswers((prev) => ({ ...prev, [questionIndex]: optionIndex }))
   271→  }
   272→
   273→  // ─── Submit quiz ──────────────────────────────────────
   274→  const handleSubmitQuiz = async () => {
   275→    if (!selectedProductId) return
   276→    setSubmitting(true)
   277→
   278→    try {
   279→      let correctCount = 0
   280→      const quizAnswersMap: Record<string, number> = {}
   281→
   282→      quizQuestions.forEach((quiz, idx) => {
   283→        const userAnswer = answers[idx]
   284→        quizAnswersMap[quiz.id] = userAnswer ?? -1
   285→        if (userAnswer === quiz.answer) {
   286→          correctCount++
   287→        }
   288→      })
   289→
   290→      const totalQuestions = quizQuestions.length
   291→      const passed = correctCount >= PASS_THRESHOLD
   292→
   293→      const result: QuizResult = {
   294→        correct: correctCount,
   295→        total: totalQuestions,
   296→        passed,
   297→        answers: { ...answers },
   298→      }
   299→      setQuizResult(result)
   300→
   301→      // Submit via service (only if logged in — otherwise just check locally)
   302→      if (user) {
   303→        const overallScore = totalQuestions > 0 ? Math.round((correctCount / totalQuestions) * 100) : 0
   304→        const res = await productLearningService.submitQuiz(
   305→          user.user_id,
   306→          selectedProductId,
   307→          quizAnswersMap,
   308→          overallScore,
   309→          passed
   310→        )
   311→
   312→        if (res.data) {
   313→          setProgress(res.data)
   314→        }
   315→      }
   316→
   317→      if (passed) {
   318→        // Mark this quiz as passed
   319→        setPassedQuizzes((prev) => ({
   320→          ...prev,
   321→          [currentStep.type === 'quiz' ? currentStep.videoIndex : -1]: true,
   322→        }))
   323→        toast.success('Quiz passed! Moving to the next lesson.')
   324→      } else {
   325→        toast.error(`You got ${correctCount}/${totalQuestions} correct. You need ${PASS_THRESHOLD} to pass. Please re-watch the video and try again.`)
   326→      }
   327→    } catch (err) {
   328→      console.error('Quiz submit failed:', err)
   329→      toast.error('Failed to submit quiz. Please try again.')
   330→    } finally {
   331→      setSubmitting(false)
   332→    }
   333→  }
   334→
   335→  // ─── Advance after passing quiz ───────────────────────
   336→  const handleQuizPassContinue = () => {
   337→    if (currentStep.type !== 'quiz') return
   338→    const nextVideoIndex = currentStep.videoIndex + 1
   339→    if (nextVideoIndex < videos.length) {
   340→      setCurrentStep({ type: 'video', videoIndex: nextVideoIndex })
   341→    } else {
   342→      // All quizzes passed — unlock the product
   343→      setCurrentStep({ type: 'completed' })
   344→    }
   345→  }
   346→
   347→  // ─── Retry after failing quiz ─────────────────────────
   348→  const handleRetryQuiz = () => {
   349→    if (currentStep.type !== 'quiz') return
   350→    // Go back to the video for this quiz
   351→    setCurrentStep({ type: 'video', videoIndex: currentStep.videoIndex })
   352→  }
   353→
   354→  // ─── Check if a step is unlocked ──────────────────────
   355→  const isStepUnlocked = (step: { type: 'video' | 'quiz'; videoIndex: number }): boolean => {
   356→    // Step 0 (Video 1) is always unlocked
   357→    if (step.videoIndex === 0 && step.type === 'video') return true
   358→    // Quiz for video 0 is unlocked if video 0 is 100% complete
   359→    if (step.videoIndex === 0 && step.type === 'quiz') {
   360→      return (progress?.video_progress?.[videos[0]?.id] ?? 0) >= 100
   361→    }
   362→    // Video N is unlocked if quiz for video N-1 is passed
   363→    if (step.type === 'video' && step.videoIndex > 0) {
   364→      return !!passedQuizzes[step.videoIndex - 1]
   365→    }
   366→    // Quiz N is unlocked if video N is 100% complete
   367→    if (step.type === 'quiz' && step.videoIndex > 0) {
   368→      const videoComplete = (progress?.video_progress?.[videos[step.videoIndex]?.id] ?? 0) >= 100
   369→      const prevQuizPassed = !!passedQuizzes[step.videoIndex - 1]
   370→      return videoComplete && prevQuizPassed
   371→    }
   372→    return false
   373→  }
   374→
   375→  // ─── Overall progress percentage ──────────────────────
   376→  const totalSteps = videos.length * 2 // each video + quiz
   377→  const completedSteps = Object.keys(passedQuizzes).filter((k) => passedQuizzes[Number(k)]).length * 2
   378→    + videos.filter((v) => (progress?.video_progress?.[v.id] ?? 0) >= 100 && !passedQuizzes[videos.indexOf(v)]).length
   379→  const overallPct = totalSteps > 0 ? Math.round((completedSteps / totalSteps) * 100) : 0
   380→
   381→  // ─── Loading state ─────────────────────────────────────
   382→  if (loading) {
   383→    return (
   384→      <div className="flex items-center justify-center min-h-[60vh]">
   385→        <div className="flex flex-col items-center gap-3">
   386→          <RefreshCw className="h-8 w-8 animate-spin" style={{ color: BRAND.green }} />
   387→          <p className="text-sm" style={{ color: BRAND.muted }}>Loading learning content...</p>
   388→        </div>
   389→      </div>
   390→    )
   391→  }
   392→
   393→  if (!product || videos.length === 0) {
   394→    return (
   395→      <div className="flex items-center justify-center min-h-[60vh]">
   396→        <Card className="max-w-md w-full" style={{ borderColor: BRAND.surface }}>
   397→          <CardContent className="p-6 text-center">
   398→            <AlertCircle className="h-10 w-10 mx-auto mb-3" style={{ color: BRAND.muted }} />
   399→            <p className="font-heading text-lg" style={{ color: BRAND.dark }}>No learning content available</p>
   400→            <p className="text-sm mt-1" style={{ color: BRAND.muted }}>This product doesn't have learning videos yet.</p>
   401→            <Button
   402→              className="mt-4"
   403→              style={{ backgroundColor: BRAND.green, color: '#fff' }}
   404→              onClick={() => navigateTo('products')}
   405→            >
   406→              <ArrowLeft className="h-4 w-4 mr-2" /> Back to Products
   407→            </Button>
   408→          </CardContent>
   409→        </Card>
   410→      </div>
   411→    )
   412→  }
   413→
   414→  // ═══════════════════════════════════════════════════════
   415→  // RENDER
   416→  // ═══════════════════════════════════════════════════════
   417→  return (
   418→    <motion.div {...fadeInUp} transition={{ duration: 0.4 }} className="max-w-4xl mx-auto px-4 py-6">
   419→      {/* ─── Header ────────────────────────────────────── */}
   420→      <div className="flex items-center gap-3 mb-4">
   421→        <Button
   422→          variant="ghost"
   423→          size="sm"
   424→          onClick={() => navigateTo('products')}
   425→          className="p-2"
   426→        >
   427→          <ArrowLeft className="h-5 w-5" style={{ color: BRAND.dark }} />
   428→        </Button>
   429→        <div className="flex-1">
   430→          <h1 className="font-heading text-xl sm:text-2xl font-bold" style={{ color: BRAND.dark }}>
   431→            Learn: {product.name}
   432→          </h1>
   433→          <p className="text-sm" style={{ color: BRAND.muted }}>
   434→            Complete each video and quiz to unlock this product
   435→          </p>
   436→        </div>
   437→        <Badge
   438→          style={{
   439→            backgroundColor: currentStep.type === 'completed' ? BRAND.lime : BRAND.surface,
   440→            color: currentStep.type === 'completed' ? BRAND.dark : BRAND.muted,
   441→          }}
   442→          className="text-xs font-medium"
   443→        >
   444→          {currentStep.type === 'completed' ? 'Unlocked' : `${overallPct}% Progress`}
   445→        </Badge>
   446→      </div>
   447→
   448→      {/* ─── Step Progress Indicator ───────────────────── */}
   449→      <Card className="mb-6" style={{ borderColor: BRAND.surface }}>
   450→        <CardContent className="p-4">
   451→          <div className="flex items-center justify-between gap-1 overflow-x-auto pb-1">
   452→            {stepList.map((step, idx) => {
   453→              const isCurrent =
   454→                currentStep.type !== 'completed' &&
   455→                step.type === currentStep.type &&
   456→                step.videoIndex === currentStep.videoIndex
   457→              const isPast = idx < currentStepIndex || currentStep.type === 'completed'
   458→              const isUnlocked = isStepUnlocked(step)
   459→              const isQuizPassed = step.type === 'quiz' && !!passedQuizzes[step.videoIndex]
   460→              const isVideoComplete =
   461→                step.type === 'video' &&
   462→                (progress?.video_progress?.[videos[step.videoIndex]?.id] ?? 0) >= 100
   463→
   464→              // Determine icon and color
   465→              let stepIcon: React.ReactNode
   466→              let bgColor: string
   467→              let iconColor: string
   468→              let textColor: string
   469→
   470→              if (isQuizPassed || (isPast && step.type === 'quiz')) {
   471→                stepIcon = <CheckCircle className="h-4 w-4" />
   472→                bgColor = BRAND.lime
   473→                iconColor = BRAND.dark
   474→                textColor = BRAND.dark
   475→              } else if (isPast && step.type === 'video') {
   476→                stepIcon = <CheckCircle className="h-4 w-4" />
   477→                bgColor = BRAND.lime
   478→                iconColor = BRAND.dark
   479→                textColor = BRAND.dark
   480→              } else if (isVideoComplete && step.type === 'video') {
   481→                stepIcon = <CheckCircle className="h-4 w-4" />
   482→                bgColor = BRAND.lime
   483→                iconColor = BRAND.dark
   484→                textColor = BRAND.dark
   485→              } else if (isCurrent) {
   486→                stepIcon =
   487→                  step.type === 'video' ? (
   488→                    <Play className="h-4 w-4" fill="white" />
   489→                  ) : (
   490→                    <FileQuestion className="h-4 w-4" />
   491→                  )
   492→                bgColor = BRAND.green
   493→                iconColor = '#fff'
   494→                textColor = BRAND.green
   495→              } else if (isUnlocked) {
   496→                stepIcon =
   497→                  step.type === 'video' ? (
   498→                    <Video className="h-4 w-4" />
   499→                  ) : (
   500→                    <FileQuestion className="h-4 w-4" />
   501→                  )
   502→                bgColor = `${BRAND.green}20`
   503→                iconColor = BRAND.green
   504→                textColor = BRAND.green
   505→              } else {
   506→                stepIcon = <Lock className="h-3.5 w-3.5" />
   507→                bgColor = BRAND.surface
   508→                iconColor = BRAND.muted
   509→                textColor = BRAND.muted
   510→              }
   511→
   512→              return (
   513→                <React.Fragment key={`${step.type}-${step.videoIndex}`}>
   514→                  {idx > 0 && (
   515→                    <div
   516→                      className="h-0.5 flex-1 min-w-[8px] max-w-[32px] rounded-full"
   517→                      style={{
   518→                        backgroundColor:
   519→                          isPast || isCurrent ? BRAND.lime : BRAND.surface,
   520→                      }}
   521→                    />
   522→                  )}
   523→                  <div className="flex flex-col items-center gap-1 min-w-[48px]">
   524→                    <div
   525→                      className="h-8 w-8 rounded-full flex items-center justify-center transition-all"
   526→                      style={{ backgroundColor: bgColor, color: iconColor }}
   527→                    >
   528→                      {stepIcon}
   529→                    </div>
   530→                    <span
   531→                      className="text-[10px] font-medium whitespace-nowrap"
