   610→   610→function TrackingTimeline({ tracking, currentStatus }: { tracking: OrderTracking[]; currentStatus: string }) {
   611→   611→  const currentStepIndex = TRACKING_STEPS.indexOf(currentStatus)
   612→   612→  const sortedTracking = [...tracking].sort((a, b) => new Date(a.tracked_at).getTime() - new Date(b.tracked_at).getTime())
   613→   613→
   614→   614→  return (
   615→   615→    <div className="relative pl-6">
   616→   616→      {/* Vertical line */}
   617→   617→      <div className="absolute left-[11px] top-0 bottom-0 w-[2px] bg-[#e3dfd8]" />
   618→   618→
   619→   619→      {sortedTracking.map((step, idx) => {
   620→   620→        const stepIndex = TRACKING_STEPS.indexOf(step.status)
   621→   621→        const isCompleted = stepIndex <= currentStepIndex && currentStepIndex >= 0
   622→   622→        const isCurrent = step.status === currentStatus
   623→   623→
   624→   624→        return (
   625→   625→          <div key={step.id} className="relative pb-4 last:pb-0">
   626→   626→            {/* Dot */}
   627→   627→            <div className={`absolute -left-[6px] top-1 w-3 h-3 rounded-full border-2 ${isCompleted ? 'bg-[#48805b] border-[#48805b]' : isCurrent ? 'bg-[#afb75d] border-[#afb75d]' : 'bg-white border-[#e3dfd8]'}`} />
   628→   628→
   629→   629→            <div className="ml-4">
   630→   630→              <div className="flex items-center gap-2">
   631→   631→                <p className={`text-sm font-semibold ${isCompleted ? 'text-[#48805b]' : isCurrent ? 'text-[#afb75d]' : 'text-[#88837b]'}`}>
   632→   632→                  {ORDER_STATUS_CONFIG[step.status]?.label || step.status}
   633→   633→                </p>
   634→   634→                {isCurrent && <Badge className="bg-[#48805b]/10 text-[#48805b] border-[#48805b]/20 text-xs">Current</Badge>}
