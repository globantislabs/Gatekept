    68→// ─── ANIMATED SECTION WRAPPER ────────────────────────────────
    69→function AnimatedSection({ children, className = '', id }: { children: React.ReactNode; className?: string; id?: string }) {
    70→  const ref = useRef(null)
    71→  const isInView = useInView(ref, { once: true, margin: '-80px' })
    72→  return (
    73→    <motion.section
    74→      ref={ref}
    75→      id={id}
    76→      initial="hidden"
    77→      animate={isInView ? 'visible' : 'hidden'}
    78→      variants={staggerContainer}
    79→      className={className}
    80→    >
    81→      {children}
    82→    </motion.section>
    83→  )
    84→}

    86→// ─── ANIMATED COUNTER ────────────────────────────────────────
    87→function AnimatedCounter({ end, suffix = '', prefix = '' }: { end: number; suffix?: string; prefix?: string }) {
    88→  const [count, setCount] = useState(0)
    89→  const ref = useRef(null)
    90→  const isInView = useInView(ref, { once: true })
    91→  const duration = 2
    92→
    93→  useEffect(() => {
    94→    if (!isInView) return
    95→    let start = 0
    96→    const step = end / (duration * 60)
    97→    const timer = setInterval(() => {
    98→      start += step
    99→      if (start >= end) { setCount(end); clearInterval(timer) }
   100→      else setCount(Math.floor(start))
   101→    }, 1000 / 60)
   102→    return () => clearInterval(timer)
   103→  }, [isInView, end, duration])
   104→
   105→  return <span ref={ref}>{prefix}{count.toLocaleString()}{suffix}</span>
   106→}
