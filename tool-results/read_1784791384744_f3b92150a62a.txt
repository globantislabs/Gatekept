    45→// ─── ANIMATED SECTION WRAPPER ────────────────────────────────
    46→function AnimatedSection({ children, className = '', id }: { children: React.ReactNode; className?: string; id?: string }) {
    47→  const ref = useRef(null)
    48→  const isInView = useInView(ref, { once: true, margin: '-80px' })
    49→  return (
    50→    <motion.section
    51→      ref={ref}
    52→      id={id}
    53→      initial="hidden"
    54→      animate={isInView ? 'visible' : 'hidden'}
    55→      variants={staggerContainer}
    56→      className={className}
    57→    >
    58→      {children}
    59→    </motion.section>
    60→  )
    61→}
