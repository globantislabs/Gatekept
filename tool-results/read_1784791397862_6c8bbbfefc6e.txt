    59→// ─── Mobile Detection ────────────────────────────────────
    60→function useIsMobile() {
    61→  const [isMobile, setIsMobile] = useState(false)
    62→  useEffect(() => {
    63→    const check = () => setIsMobile(window.innerWidth < 768)
    64→    check()
    65→    window.addEventListener('resize', check)
    66→    return () => window.removeEventListener('resize', check)
    67→  }, [])
    68→  return isMobile
    69→}
