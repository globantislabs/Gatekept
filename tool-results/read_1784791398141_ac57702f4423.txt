    55→// ─── Helper: Parse Highlights ───────────────────────────────
    56→function parseHighlights(highlights: string | null | undefined): string[] {
    57→  if (!highlights) return []
    58→  return highlights.split(',').map(h => h.trim()).filter(Boolean)
    59→}
    60→
    61→// ─── Helper: Parse Ingredients ──────────────────────────────
    62→function parseIngredients(ingredients: string | null | undefined): string[] {
    63→  if (!ingredients) return []
    64→  return ingredients.split(',').map(i => i.trim()).filter(Boolean)
    65→}
    66→
    67→// ─── Helper: Discount Calculation ───────────────────────────
    68→function calculateDiscount(price: number, mrp: number | null | undefined): { discountPercent: number; savings: number } | null {
    69→  if (!mrp || mrp <= price) return null
    70→  const savings = mrp - price
    71→  const discountPercent = Math.round((savings / mrp) * 100)
    72→  return { discountPercent, savings }
    73→}
    74→
    75→// ─── Helper: Video Progress Percentage ──────────────────────
    76→function getVideoProgressPercent(progress: ProductLearningProgress | null): number {
    77→  if (!progress) return 0
    78→  const vp = progress.video_progress
    79→  const total = Object.keys(vp).length
    80→  if (total === 0) return 0
    81→  const watched = Object.values(vp).reduce((sum, pct) => sum + (pct >= 0.9 ? 1 : 0), 0)
    82→  return Math.round((watched / total) * 100)
    83→}

    85→// ─── Helper: Learning Status ────────────────────────────────

    87→
    88→function getLearningStatus(progress: ProductLearningProgress | null): LearningStatus {
    89→  if (!progress) return 'NOT_STARTED'
    90→  return progress.status as LearningStatus
    91→}
