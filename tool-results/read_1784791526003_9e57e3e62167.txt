   905→   905→// Shows past, current, and future deliveries for a subscription
   906→   906→function DeliveryScheduleTimeline({ subscription }: { subscription: Subscription }) {
   907→   907→  const { start_date, frequency_days, completed_cycles, total_cycles, next_delivery, status } = subscription
   908→   908→
   909→   909→  // Generate delivery dates
   910→   910→  const startDate = new Date(start_date)
   911→   911→  const deliveries: { date: Date; cycle: number; type: 'past' | 'next' | 'future' }[] = []
   912→   912→
   913→   913→  for (let cycle = 1; cycle <= total_cycles; cycle++) {
   914→   914→    const deliveryDate = new Date(startDate.getTime() + (cycle - 1) * frequency_days * 24 * 60 * 60 * 1000)
   915→   915→    const type = cycle <= completed_cycles ? 'past' : cycle === completed_cycles + 1 ? 'next' : 'future'
   916→   916→    deliveries.push({ date: deliveryDate, cycle, type })
   917→   917→  }
   918→   918→
   919→   919→  // Show max 3 past, 1 next, 3 future
   920→   920→  const pastDeliveries = deliveries.filter(d => d.type === 'past').slice(-3)
   921→   921→  const nextDelivery = deliveries.find(d => d.type === 'next')
   922→   922→  const futureDeliveries = deliveries.filter(d => d.type === 'future').slice(0, 3)
   923→   923→
   924→   924→  const displayItems = [...pastDeliveries, nextDelivery, ...futureDeliveries].filter(Boolean)
   925→   925→
   926→   926→  return (
   927→   927→    <div className="p-4 rounded-xl bg-[#f4f3f0] border border-[#e3dfd8]">
   928→   928→      {/* ── Cycle progress bar ── */}

   942→   942→        {displayItems.map((item, idx) => {
   943→   943→          const isNext = item.type === 'next'
   944→   944→          const isPast = item.type === 'past'
   945→   945→          const isFuture = item.type === 'future'

   948→   948→          return (
