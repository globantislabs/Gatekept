   115→// ─── OTP Action Gate ────────────────────────────────────
   116→// Tracks what action is pending OTP verification
   117→type OtpAction =
   118→  | { type: 'CANCEL_ORDER'; orderId: string }
   119→  | { type: 'CANCEL_SUB'; subId: string }
   120→  | { type: 'PAUSE_SUB'; subId: string }
   121→  | { type: 'RESUME_SUB'; subId: string }
   122→  | { type: 'MODIFY_ADDRESS'; orderId: string }
   123→
   124→function getOtpPurpose(action: OtpAction): OtpPurpose {
   125→  switch (action.type) {
   126→    case 'CANCEL_ORDER': return 'CANCEL_ORDER'
   127→    case 'CANCEL_SUB': return 'CANCEL_SUB'
   128→    case 'PAUSE_SUB': return 'PAUSE_SUB'
   129→    case 'RESUME_SUB': return 'RESUME_SUB'
   130→    case 'MODIFY_ADDRESS': return 'MODIFY_ADDRESS'
   131→  }
   132→}
   133→
   134→function getOtpReferenceId(action: OtpAction): string {
   135→  switch (action.type) {
   136→    case 'CANCEL_ORDER': return action.orderId
   137→    case 'CANCEL_SUB': return action.subId
   138→    case 'PAUSE_SUB': return action.subId
   139→    case 'RESUME_SUB': return action.subId
   140→    case 'MODIFY_ADDRESS': return action.orderId
   141→  }
   142→}
   143→
   144→function getOtpLabel(action: OtpAction): string {
   145→  switch (action.type) {
   146→    case 'CANCEL_ORDER': return 'Cancel Order'
   147→    case 'CANCEL_SUB': return 'Cancel Subscription'
   148→    case 'PAUSE_SUB': return 'Pause Subscription'
   149→    case 'RESUME_SUB': return 'Resume Subscription'
   150→    case 'MODIFY_ADDRESS': return 'Modify Shipping Address'
   151→  }
   152→}

   904→// ─── Delivery Schedule Timeline ──────────────────────────
   905→// Shows past, current, and future deliveries for a subscription
   906→function DeliveryScheduleTimeline({ subscription }: { subscription: Subscription }) {
   907→  const { start_date, frequency_days, completed_cycles, total_cycles, next_delivery, status } = subscription
   908→
   909→  // Generate delivery dates
   910→  const startDate = new Date(start_date)
   911→  const deliveries: { date: Date; cycle: number; type: 'past' | 'next' | 'future' }[] = []
   912→
   913→  for (let cycle = 1; cycle <= total_cycles; cycle++) {
   914→    const deliveryDate = new Date(startDate.getTime() + (cycle - 1) * frequency_days * 24 * 60 * 60 * 1000)
   915→    const type = cycle <= completed_cycles ? 'past' : cycle === completed_cycles + 1 ? 'next' : 'future'
   916→    deliveries.push({ date: deliveryDate, cycle, type })
   917→  }
   918→
   919→  // Show max 3 past, 1 next, 3 future
   920→  const pastDeliveries = deliveries.filter(d => d.type === 'past').slice(-3)
   921→  const nextDelivery = deliveries.find(d => d.type === 'next')
   922→  const futureDeliveries = deliveries.filter(d => d.type === 'future').slice(0, 3)
   923→
   924→  const displayItems = [...pastDeliveries, nextDelivery, ...futureDeliveries].filter(Boolean)
   925→
   926→  return (
   927→    <div className="p-4 rounded-xl bg-[#f4f3f0] border border-[#e3dfd8]">
   928→      {/* ── Cycle progress bar ── */}
   929→      <div className="mb-3">
   930→        <div className="flex items-center justify-between mb-1">
   931→          <span className="text-xs text-[#88837b]">Delivery Progress</span>
   932→          <span className="text-xs font-semibold text-[#1f1e1c]">{completed_cycles}/{total_cycles} cycles</span>
   933→        </div>
   934→        <Progress
   935→          value={(completed_cycles / total_cycles) * 100}
   936→          className="h-2 bg-[#e3dfd8]"
   937→        />
   938→      </div>
   939→
   940→      {/* ── Timeline entries ── */}
   941→      <div className="space-y-2">
   942→        {displayItems.map((item, idx) => {
   943→          const isNext = item.type === 'next'
   944→          const isPast = item.type === 'past'
   945→          const isFuture = item.type === 'future'
   946→          const isPaused = status === 'PAUSED' && (isNext || isFuture)
   947→
   948→          return (
   949→            <motion.div
   950→              key={`${item.cycle}-${idx}`}
   951→              initial={{ opacity: 0, y: 8 }}

   955→                isNext ? 'bg-[#48805b]/10 border border-[#48805b]/20' :
   956→                isFuture ? 'bg-[#f4f3f0]' :
   957→                'bg-[#e3dfd8]/30'
   958→              }`}
   959→            >
   960→              {/* Icon */}
   961→              <div className={`w-6 h-6 rounded-full flex items-center justify-center shrink-0 ${
   962→                isPast ? 'bg-[#48805b] text-white' :
   963→                isNext ? 'bg-[#afb75d] text-white' :
   964→                'bg-[#e3dfd8] text-[#88837b]'
   965→              }`}>
   966→                {isPast ? <CheckCircle className="w-3.5 h-3.5" /> :
   967→                 isNext ? <Truck className="w-3.5 h-3.5" /> :
